<script>
const MAX_DOING = 2;
const board = document.querySelector('.board');
function updateCounters(){
  const todo = document.querySelectorAll('.col[data-col="todo"] .card').length;
  const doing = document.querySelectorAll('.col[data-col="doing"] .card').length;
  const done = document.querySelectorAll('.col[data-col="done"] .card').length;
  document.getElementById('cont-todo').textContent = todo;
  document.getElementById('cont-doing').textContent = doing;
  document.getElementById('cont-done').textContent = done;
}
function moveCard(card, dir){
  const cardsContainer = card.parentElement.parentElement; // .cards
  const colName = cardsContainer.parentElement.dataset.col;
  const cols = ['todo','doing','done'];
  let idx = cols.indexOf(colName);
  let newIdx = idx + dir;
  if(newIdx < 0 || newIdx >= cols.length) return; // cannot move
  // WIP rule
  if(cols[newIdx] === 'doing'){
    const doingCards = document.querySelectorAll('.col[data-col="doing"] .card');
    if(doingCards.length >= MAX_DOING) return;
  }
  const newCol = document.querySelector('.col[data-col="'+cols[newIdx]+'"] .cards');
  newCol.appendChild(card);
  updateCounters();
}
board.addEventListener('click', function(e){
  if(e.target.classList.contains('mas')){
    const card = e.target.closest('.card');
    moveCard(card, 1);
  } else if(e.target.classList.contains('menos')){
    const card = e.target.closest('.card');
    moveCard(card, -1);
  }
});
updateCounters();
</script>
