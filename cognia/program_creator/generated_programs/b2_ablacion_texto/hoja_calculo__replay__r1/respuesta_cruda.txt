<script>
(() => {
  const cells = Array.from(document.querySelectorAll('.celda'));
  const cellMap = {};
  cells.forEach(c => cellMap[c.dataset.ref] = c);

  // Helper to parse and evaluate a formula string
  function evaluateFormula(formula, stack = []) {
    // Replace references with evaluated values
    const expr = formula.replace(/[A-C][1-3]/g, ref => {
      if (stack.includes(ref)) return 'CIRC';
      const cell = cellMap[ref];
      if (!cell) return '0';
      const val = evaluateCell(ref, [...stack, ref]);
      return val === 'CIRC' ? 'CIRC' : val;
    });
    if (expr.includes('CIRC')) return 'CIRC';
    try {
      // Safe eval: only numbers, +, -, parentheses
      const result = eval(expr);
      return isNaN(result) ? 'ERR' : result;
    } catch (e) {
      return 'ERR';
    }
  }

  function evaluateCell(ref, stack) {
    const cell = cellMap[ref];
    if (!cell) return 0;
    const formula = cell.dataset.formula;
    if (!formula) {
      const val = parseFloat(cell.value);
      return isNaN(val) ? 0 : val;
    }
    return evaluateFormula(formula, stack);
  }

  function recalcAll() {
    cells.forEach(cell => {
      const formula = cell.dataset.formula;
      if (formula) {
        const result = evaluateFormula(formula, [cell.dataset.ref]);
        if (result === 'CIRC') {
          cell.value = 'CIRC';
        } else if (result === 'ERR') {
          cell.value = 'ERR';
        } else {
          cell.value = Number(result).toLocaleString(undefined, {maximumFractionDigits: 2});
        }
      }
    });
    document.getElementById('status').textContent = 'Recalculado';
  }

  cells.forEach(cell => {
    cell.addEventListener('input', e => {
      const val = e.target.value.trim();
      if (val.startsWith('=')) {
        cell.dataset.formula = val.slice(1);
      } else {
        delete cell.dataset.formula;
      }
      recalcAll();
    });
  });

  // Initial calculation
  recalcAll();
})();
</script>
