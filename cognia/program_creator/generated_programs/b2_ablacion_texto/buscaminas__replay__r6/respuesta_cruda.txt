<script>
const GRID_SIZE = 5;
const MINES = [6,12,18];
const cells = document.querySelectorAll('.c');
const cellCount = GRID_SIZE * GRID_SIZE;
const adjacentMines = new Array(cellCount).fill(0);
const opened = new Set();
const flagged = new Set();
const estado = document.getElementById('estado');
const resetBtn = document.getElementById('reset');

function computeAdjacentMines() {
  for (let i=0;i<cellCount;i++) {
    if (MINES.includes(i)) { adjacentMines[i] = -1; continue; }
    let count=0;
    const neighbors = getNeighbors(i);
    for (const n of neighbors) if (MINES.includes(n)) count++;
    adjacentMines[i] = count;
  }
}

function getNeighbors(i) {
  const r = Math.floor(i/GRID_SIZE);
  const c = i%GRID_SIZE;
  const neighbors=[];
  for (let dr=-1;dr<=1;dr++) {
    for (let dc=-1;dc<=1;dc++) {
      if (dr===0 && dc===0) continue;
      const nr=r+dr, nc=c+dc;
      if (nr>=0 && nr<GRID_SIZE && nc>=0 && nc<GRID_SIZE) {
        neighbors.push(nr*GRID_SIZE+nc);
      }
    }
  }
  return neighbors;
}

function openCell(i) {
  if (opened.has(i) || flagged.has(i)) return;
  opened.add(i);
  const cell = cells[i];
  cell.classList.add('abierta');
  if (MINES.includes(i)) {
    cell.classList.add('mine');
    estado.textContent='perdiste';
    return;
  }
  const count = adjacentMines[i];
  if (count>0) { cell.textContent = count; }
  else { cell.textContent=''; openNeighborsCascade(i); }
}

function openNeighborsCascade(start) {
  const queue=[start];
  const visited=new Set();
  while(queue.length){
    const idx=queue.shift();
    if(visited.has(idx)) continue;
    visited.add(idx);
    const neighbors=getNeighbors(idx);
    for(const n of neighbors){
      if(opened.has(n) || flagged.has(n) || MINES.includes(n)) continue;
      opened.add(n);
      const cell=cells[n];
      cell.classList.add('abierta');
      const cnt=adjacentMines[n];
      if(cnt>0){ cell.textContent=cnt; }
      else { cell.textContent=''; queue.push(n); }
    }
  }
}

function toggleFlag(i) {
  if (opened.has(i)) return;
  const cell=cells[i];
  if(flagged.has(i)){
    flagged.delete(i);
    cell.classList.remove('bandera');
    cell.textContent='';
  } else {
    flagged.add(i);
    cell.classList.add('bandera');
    cell.textContent='';
  }
}

function resetGame(){
  opened.clear();
  flagged.clear();
  estado.textContent='';
  cells.forEach(cell=>{
    cell.classList.remove('abierta','bandera','mine');
    cell.textContent='';
  });
}

cells.forEach(cell=>{
  const i=parseInt(cell.dataset.i);
  cell.addEventListener('click',()=>{ openCell(i); });
  cell.addEventListener('contextmenu',e=>{
    e.preventDefault();
    toggleFlag(i);
  });
});

resetBtn.addEventListener('click',resetGame);

computeAdjacentMines();
</script>
