<script>
  // Helper to format currency
  function formatCurrency(v){return v.toLocaleString('es-ES',{style:'currency',currency:'EUR'});}
  // Initialize product prices display
  document.querySelectorAll('.prod .price').forEach(el=>{
    const price = parseFloat(el.parentElement.parentElement.dataset.precio);
    el.textContent = formatCurrency(price);
  });
  const cartLines = document.getElementById('cart-lines');
  const totalEl = document.getElementById('total');
  const addButtons = document.querySelectorAll('.add');
  const cartState = {}; // {id:{qty,price,stock,name}}
  function updateTotal(){
    let sum=0;
    Object.values(cartState).forEach(item=>sum+=item.qty*item.price);
    totalEl.textContent = formatCurrency(sum);
  }
  function createLine(id){
    const prodEl = document.querySelector(`.prod[data-id="${id}"]`);
    const name = prodEl.dataset.name;
    const price = parseFloat(prodEl.dataset.precio);
    const stock = parseInt(prodEl.dataset.stock,10);
    const line = document.createElement('div');
    line.className='linea';
    line.dataset.id=id;
    line.innerHTML = `
      <span class="name">${name}</span>
      <span class="cant">1</span>
      <span class="line-total">${formatCurrency(price)}</span>
      <button class="quitar" data-id="${id}">Quitar</button>
    `;
    cartLines.appendChild(line);
    cartState[id]={qty:1,price,stock,name};
    // disable add if stock reached
    if(cartState[id].qty>=stock) document.querySelector(`.add[data-id="${id}"]`).disabled=true;
    // attach quitar handler
    line.querySelector('.quitar').addEventListener('click',()=>removeOne(id));
  }
  function addOne(id){
    const prodEl = document.querySelector(`.prod[data-id="${id}"]`);
    const price = parseFloat(prodEl.dataset.precio);
    const stock = parseInt(prodEl.dataset.stock,10);
    if(!cartState[id]){
      createLine(id);
    }else{
      if(cartState[id].qty<stock){
        cartState[id].qty++;
        const line = document.querySelector(`.linea[data-id="${id}"]`);
        line.querySelector('.cant').textContent=cartState[id].qty;
        line.querySelector('.line-total').textContent=formatCurrency(cartState[id].qty*price);
        if(cartState[id].qty>=stock) document.querySelector(`.add[data-id="${id}"]`).disabled=true;
      }
    }
    updateTotal();
  }
  function removeOne(id){
    if(!cartState[id]) return;
    cartState[id].qty--;
    if(cartState[id].qty===0){
      const line=document.querySelector(`.linea[data-id="${id}"]`);
      line.remove();
      delete cartState[id];
      document.querySelector(`.add[data-id="${id}"]`).disabled=false;
    }else{
      const line=document.querySelector(`.linea[data-id="${id}"]`);
      line.querySelector('.cant').textContent=cartState[id].qty;
      line.querySelector('.line-total').textContent=formatCurrency(cartState[id].qty*cartState[id].price);
      document.querySelector(`.add[data-id="${id}"]`).disabled=false;
    }
    updateTotal();
  }
  addButtons.forEach(btn=>{
    btn.addEventListener('click',()=>addOne(btn.dataset.id));
  });
</script>
