<script>
  // Constants
  const GRID_SIZE = 5;
  const TOTAL_CELLS = GRID_SIZE * GRID_SIZE;
  const MINES = new Set([6,12,18]);

  // State
  const opened = Array(TOTAL_CELLS).fill(false);
  const flagged = Array(TOTAL_CELLS).fill(false);
  let gameOver = false;

  // Precompute neighbor indices
  const neighbors = Array.from({length:TOTAL_CELLS}, (_,i)=> {
    const r = Math.floor(i/GRID_SIZE);
    const c = i%GRID_SIZE;
    const list=[];
    for(let dr=-1;dr<=1;dr++){
      for(let dc=-1;dc<=1;dc++){
        if(dr===0 && dc===0) continue;
        const nr=r+dr, nc=c+dc;
        if(nr>=0 && nr<GRID_SIZE && nc>=0 && nc<GRID_SIZE){
          list.push(nr*GRID_SIZE+nc);
        }
      }
    }
    return list;
  });

  // Precompute adjacent mine counts
  const adjCount = neighbors.map(list=>list.reduce((s,i)=>s+(MINES.has(i)?1:0),0));

  const gridEl = document.getElementById('grid');
  const estadoEl = document.getElementById('estado');
  const resetBtn = document.getElementById('reset');

  // Build grid
  for(let i=0;i<TOTAL_CELLS;i++){
    const div = document.createElement('div');
    div.className='c';
    div.dataset.i=i;
    div.addEventListener('click',()=>handleClick(i));
    div.addEventListener('contextmenu',e=>{
      e.preventDefault();
      toggleFlag(i);
    });
    gridEl.appendChild(div);
  }

  function handleClick(i){
    if(gameOver || flagged[i] || opened[i]) return;
    openCell(i);
  }

  function toggleFlag(i){
    if(gameOver || opened[i]) return;
    flagged[i] = !flagged[i];
    const cell = gridEl.children[i];
    if(flagged[i]) cell.classList.add('bandera');
    else cell.classList.remove('bandera');
  }

  function openCell(i){
    if(opened[i] || flagged[i] || gameOver) return;
    opened[i] = true;
    const cell = gridEl.children[i];
    if(MINES.has(i)){
      cell.classList.add('mina');
      estadoEl.textContent='perdiste';
      gameOver = true;
      return;
    }
    const count = adjCount[i];
    if(count>0){
      cell.textContent=count;
    }
    cell.classList.add('abierta');
    if(count===0){
      neighbors[i].forEach(j=>openCell(j));
    }
  }

  resetBtn.addEventListener('click',resetGame);

  function resetGame(){
    opened.fill(false);
    flagged.fill(false);
    gameOver=false;
    estadoEl.textContent='';
    for(let i=0;i<TOTAL_CELLS;i++){
      const cell = gridEl.children[i];
      cell.className='c';
      cell.textContent='';
    }
  }
</script>
