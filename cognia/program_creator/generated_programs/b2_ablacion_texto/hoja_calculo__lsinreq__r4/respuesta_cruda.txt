<script>
document.addEventListener('DOMContentLoaded',()=>{const cells=document.querySelectorAll('.celda');const formulas={};const values={};
function evalCell(ref,stack){if(stack.includes(ref))return 'CIRC';const f=formulas[ref];if(!f){const v=values[ref];return v?parseFloat(v)||0:0;}
stack.push(ref);let expr=f.slice(1).replace(/[A-C][1-3]/g,r=>{const val=evalCell(r,stack);return val==='CIRC'?val:val||0;});if(expr.includes('CIRC')){stack.pop();return 'CIRC';}
try{const res=Function('return '+expr)();stack.pop();return res;}catch{stack.pop();return 'CIRC';}}
function recalc(){cells.forEach(inp=>{const ref=inp.dataset.ref;const val=inp.value;if(val.startsWith('=')){formulas[ref]=val;values[ref]=null;}else{formulas[ref]=null;values[ref]=val;}}
cells.forEach(inp=>{const ref=inp.dataset.ref;const f=formulas[ref];if(!f){inp.value=values[ref]||'';inp.classList.remove('formula');}else{const res=evalCell(ref,[]);if(res==='CIRC'){inp.value='CIRC';inp.classList.remove('formula');}else{inp.value=parseFloat(res).toLocaleString('es-ES',{minimumFractionDigits:2,maximumFractionDigits:2});inp.classList.add('formula');}}});}
cells.forEach(inp=>inp.addEventListener('input',recalc));
recalc();});
</script>
