<script>
(() => {
  const inputs = document.querySelectorAll('.celda');
  inputs.forEach(cell => cell.addEventListener('input', onInput));

  function onInput(e) {
    const cell = e.target;
    const val = cell.value;
    if (val.startsWith('=')) {
      cell.dataset.formula = val;
      cell.classList.add('formula');
    } else {
      delete cell.dataset.formula;
      cell.classList.remove('formula');
    }
    recalcAll();
  }

  function recalcAll() {
    const cells = document.querySelectorAll('.celda');
    cells.forEach(cell => {
      if (cell.dataset.formula !== undefined) {
        const result = evaluateCell(cell, new Set());
        if (result === 'CIRC') {
          cell.value = 'CIRC';
        } else {
          const num = Number(result);
          cell.value = num.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
        }
      }
    });
  }

  function evaluateCell(cell, visited) {
    const ref = cell.dataset.ref;
    if (visited.has(ref)) return 'CIRC';
    visited.add(ref);
    if (!cell.dataset.formula) {
      const val = parseFloat(cell.value);
      visited.delete(ref);
      return isNaN(val) ? 0 : val;
    }
    let expr = cell.dataset.formula.slice(1);
    const refs = expr.match(/[A-C][1-3]/g) || [];
    let evaluatedExpr = expr;
    for (const r of refs) {
      const rCell = document.querySelector(`[data-ref="${r}"]`);
      const v = evaluateCell(rCell, visited);
      if (v === 'CIRC') {
        visited.delete(ref);
        return 'CIRC';
      }
      evaluatedExpr = evaluatedExpr.replaceAll(r, v);
    }
    let result;
    try {
      result = Function('return ' + evaluatedExpr)();
    } catch (e) {
      result = 0;
    }
    visited.delete(ref);
    return result;
  }

  recalcAll();
})();
</script>
