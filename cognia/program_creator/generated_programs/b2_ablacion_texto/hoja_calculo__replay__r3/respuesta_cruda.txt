<script>
  // Formatea números con separador de miles y 2 decimales
  function formatNumber(num){
    if(typeof num !== 'number' || isNaN(num)) return num;
    return num.toLocaleString('es-ES',{minimumFractionDigits:2,maximumFractionDigits:2});
  }

  // Evalúa una celda recursivamente, detectando ciclos
  function evaluateCell(cell, visited){
    if(visited.has(cell)) return 'CIRC';
    visited.add(cell);
    const formula = cell.dataset.formula;
    if(!formula){
      const val = cell.dataset.value;
      const num = Number(val);
      return isNaN(num)?0:num;
    }
    let expr = formula.slice(1); // quitar '='
    expr = expr.replace(/[A-C][1-3]/g, ref=>{
      const refCell = document.querySelector(`[data-ref="${ref}"]`);
      if(!refCell) return '0';
      const val = evaluateCell(refCell, visited);
      if(val==='CIRC') return 'CIRC';
      return val;
    });
    if(expr.includes('CIRC')) return 'CIRC';
    try{
      const result = eval(expr);
      return result;
    }catch(e){
      return 'CIRC';
    }
  }

  // Recalcula todas las celdas con fórmulas y actualiza valores mostrados
  function recomputeAll(){
    const cells = document.querySelectorAll('.celda');
    cells.forEach(cell=>{
      if(cell.dataset.formula){
        const val = evaluateCell(cell, new Set());
        if(val==='CIRC'){
          cell.dataset.value='CIRC';
          cell.value='CIRC';
        }else{
          cell.dataset.value=val;
          cell.value=formatNumber(val);
        }
      }else{
        const val = cell.dataset.value;
        const num = Number(val);
        cell.value=formatNumber(num);
      }
    });
  }

  // Maneja cambios de usuario
  function onInput(e){
    const cell=e.target;
    const val=cell.value.trim();
    if(val.startsWith('=')){
      cell.dataset.formula=val;
      cell.dataset.value='';
    }else{
      cell.dataset.formula='';
      cell.dataset.value=val;
    }
    recomputeAll();
  }

  // Inicializa listeners y calcula valores iniciales
  document.querySelectorAll('.celda').forEach(cell=>cell.addEventListener('input',onInput));
  recomputeAll();
</script>
