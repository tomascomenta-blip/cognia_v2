<script>
document.addEventListener('DOMContentLoaded', function(){
  const products = new Map();
  const cart = new Map();
  const cartLines = document.querySelector('.cart-lines');
  const totalSpan = document.getElementById('total');
  // Initialize product data and display price/stock
  document.querySelectorAll('.prod').forEach(prodEl=>{
    const id = prodEl.dataset.id;
    const precio = parseFloat(prodEl.dataset.precio);
    const stock = parseInt(prodEl.dataset.stock,10);
    products.set(id,{precio,stock});
    prodEl.querySelector('.precio').textContent = formatCurrency(precio);
    prodEl.querySelector('.stock').textContent = stock;
  });
  // Add button click
  document.querySelectorAll('.add').forEach(btn=>{
    btn.addEventListener('click',function(){
      const id = this.dataset.id;
      addToCart(id);
    });
  });
  // Remove button via delegation
  cartLines.addEventListener('click',function(e){
    if(e.target.matches('.quitar')){
      const id = e.target.dataset.id;
      removeFromCart(id);
    }
  });
  function addToCart(id){
    const prod = products.get(id);
    const current = cart.get(id)||0;
    if(current < prod.stock){
      cart.set(id,current+1);
      updateLine(id);
      updateTotal();
      updateAddButton(id);
    }
  }
  function removeFromCart(id){
    const current = cart.get(id)||0;
    if(current>0){
      if(current===1){
        cart.delete(id);
        removeLine(id);
      }else{
        cart.set(id,current-1);
        updateLine(id);
      }
      updateTotal();
      updateAddButton(id);
    }
  }
  function updateLine(id){
    const qty = cart.get(id);
    let lineEl = cartLines.querySelector(`.linea[data-id="${id}"]`);
    if(!lineEl){
      lineEl = document.createElement('div');
      lineEl.className='linea';
      lineEl.dataset.id=id;
      lineEl.innerHTML = `<span class="cant">${qty}</span> <button class="quitar" data-id="${id}">Eliminar</button>`;
      cartLines.appendChild(lineEl);
    }else{
      lineEl.querySelector('.cant').textContent = qty;
    }
  }
  function removeLine(id){
    const lineEl = cartLines.querySelector(`.linea[data-id="${id}"]`);
    if(lineEl) lineEl.remove();
  }
  function updateTotal(){
    let sum=0;
    cart.forEach((qty,id)=>{
      sum += qty*products.get(id).precio;
    });
    totalSpan.textContent = formatCurrency(sum);
  }
  function updateAddButton(id){
    const prod = products.get(id);
    const qty = cart.get(id)||0;
    const btn = document.querySelector(`.add[data-id="${id}"]`);
    if(qty>=prod.stock){
      btn.disabled=true;
    }else{
      btn.disabled=false;
    }
  }
  function formatCurrency(val){
    return val.toLocaleString('es-ES',{style:'currency',currency:'EUR',minimumFractionDigits:2,maximumFractionDigits:2});
  }
});
</script>
