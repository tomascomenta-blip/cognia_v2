<script>
const cells = {};

function init() {
  const inputs = document.querySelectorAll('.celda');
  inputs.forEach(input => {
    const ref = input.dataset.ref;
    cells[ref] = input;
    // initial formula detection
    if (input.value.startsWith('=')) {
      input.dataset.formula = input.value;
    } else {
      input.dataset.formula = '';
    }
    input.addEventListener('input', onInput);
    input.addEventListener('focus', () => input.classList.add('active'));
    input.addEventListener('blur', () => input.classList.remove('active'));
  });
  recalcAll();
}

function onInput(e) {
  const input = e.target;
  if (input.value.startsWith('=')) {
    input.dataset.formula = input.value;
  } else {
    input.dataset.formula = '';
  }
  recalcAll();
}

function recalcAll() {
  const status = document.getElementById('status');
  status.textContent = 'Recalculando...';
  Object.keys(cells).forEach(ref => {
    const cell = cells[ref];
    if (cell.dataset.formula) {
      const val = evaluateCell(ref, []);
      if (val === 'CIRC') {
        cell.value = 'CIRC';
      } else if (typeof val === 'number') {
        cell.value = val.toLocaleString('es-ES', {minimumFractionDigits: 2, maximumFractionDigits: 2});
      } else {
        cell.value = val;
      }
    }
  });
  status.textContent = 'Listo';
}

function evaluateCell(ref, stack) {
  if (stack.includes(ref)) return 'CIRC';
  const cell = cells[ref];
  const formula = cell.dataset.formula;
  if (!formula) return cell.value;
  stack.push(ref);
  let expr = formula.slice(1); // remove '='
  expr = expr.replace(/[A-C][1-3]/g, match => {
    const val = evaluateCell(match, stack);
    if (val === 'CIRC') return 'CIRC';
    const num = parseFloat(val);
    return isNaN(num) ? 0 : num;
  });
  if (expr.includes('CIRC')) {
    stack.pop();
    return 'CIRC';
  }
  let result;
  try {
    result = eval(expr);
  } catch (e) {
    result = '';
  }
  stack.pop();
  return result;
}

document.addEventListener('DOMContentLoaded', init);
</script>
