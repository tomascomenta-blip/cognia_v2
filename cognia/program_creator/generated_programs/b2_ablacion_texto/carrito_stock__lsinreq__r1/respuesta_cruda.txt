Title: Carrito de compra con STOCK  
Description: Una página de carrito de compra con 3 productos, manejo de stock y total.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>

<meta name="viewport" content="width=device-width, initial-scale=1">
<style>
  body{
    margin:0;
    font-family:Arial,Helvetica,sans-serif;

  }
  .container{
    display:flex;
    flex-wrap:wrap;
    gap:20px;

  }
  .products, .cart{
    background:#fff;
    border-radius:8px;

    .products,.cart{flex:1 1 100%;}

<script>
document.addEventListener('DOMContentLoaded', function(){
  const cart = {};

  // Inicializar precios y stock en la vista
  document.querySelectorAll('.prod').forEach(prod=>{
    const precio = parseFloat(prod.dataset.precio);
    const stock = parseInt(prod.dataset.stock);
    prod.querySelector('.precio').textContent = precio.toLocaleString('es-ES',{style:'currency',currency:'EUR'});
    prod.querySelector('.stock').textContent = 'Stock: ' + stock;
  });

  function recalcTotal(){
    let total = 0;
    for(const id in cart){
      const prod = document.querySelector(`[data-id="${id}"]`);
      const price = parseFloat(prod.dataset.precio);
      total += cart[id] * price;
    }
    document.getElementById('total').textContent = total.toLocaleString('es-ES',{style:'currency',currency:'EUR'});
  }

  function updateAddButton(id){
    const prod = document.querySelector(`[data-id="${id}"]`);
    const stock = parseInt(prod.dataset.stock);
    const qty = cart[id] || 0;
    const btn = prod.querySelector('.add');
    btn.disabled = qty >= stock;
  }

  function createLine(id){
    const prod = document.querySelector(`[data-id="${id}"]`);
    const line = document.createElement('div');
    line.className = 'linea';
    line.dataset.id = id;
    const name = document.createElement('span');
    name.className = 'name';
    name.textContent = prod.querySelector('.name').textContent;
    const cant = document.createElement('span');
    cant.className = 'cant';
    cant.textContent = '1';
    const btn = document.createElement('button');
    btn.className = 'quitar';
    btn.dataset.id = id;
    btn.textContent = 'Eliminar';
    line.appendChild(name);
    line.appendChild(cant);
    line.appendChild(btn);
    document.getElementById('cart-lines').appendChild(line);
  }

  function updateLine(id){
    const line = document.querySelector(`.linea[data-id="${id}"]`);
    if(line){
      line.querySelector('.cant').textContent = cart[id];
    }
  }

  function handleAdd(id){
    const prod = document.querySelector(`[data-id="${id}"]`);
    const stock = parseInt(prod.dataset.stock);
    const qty = cart[id] || 0;
    if(qty < stock){
      cart[id] = qty + 1;
      if(cart[id] === 1){
        createLine(id);
      }else{
        updateLine(id);
      }
      updateAddButton(id);
      recalcTotal();
    }
  }

  function handleRemove(id){
    const qty = cart[id] || 0;
    if(qty > 0){
      cart[id] = qty - 1;
      if(cart[id] === 0){
        const line = document.querySelector(`.linea[data-id="${id}"]`);
        if(line) line.remove();
        delete cart[id];
      }else{
        updateLine(id);
      }
      updateAddButton(id);
      recalcTotal();
    }
  }

  document.addEventListener('click', function(e){
    if(e.target.matches('.add')){
      handleAdd(e.target.dataset.id);
    }else if(e.target.matches('.quitar')){
      handleRemove(e.target.dataset.id);
    }
  });
});
</script>
</body>
</html>
```
