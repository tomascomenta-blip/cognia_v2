<script>
(() => {
  const REF_REGEX = /[A-C][1-3]/g;
  const cells = {};          // ref -> input element
  const dependenciesMap = {}; // ref -> array of refs it depends on
  const dependentsMap = {};   // ref -> array of refs that depend on it

  function getCell(ref) { return cells[ref]; }

  function updateDependencies(ref) {
    const cell = getCell(ref);
    const oldDeps = dependenciesMap[ref] || [];
    oldDeps.forEach(dep => {
      const list = dependentsMap[dep];
      if (list) dependentsMap[dep] = list.filter(r => r !== ref);
    });
    let newDeps = [];
    if (cell.dataset.formula) {
      const matches = cell.dataset.formula.match(REF_REGEX);
      if (matches) newDeps = [...new Set(matches)];
    }
    dependenciesMap[ref] = newDeps;
    newDeps.forEach(dep => {
      if (!dependentsMap[dep]) dependentsMap[dep] = [];
      if (!dependentsMap[dep].includes(ref)) dependentsMap[dep].push(ref);
    });
  }

  function evaluateCell(ref, visited) {
    const cell = getCell(ref);
    if (visited.has(ref)) {
      visited.forEach(r => {
        const c = getCell(r);
        c.value = 'CIRC';
        c.classList.remove('formula');
        delete c.dataset.formula;
      });
      return NaN;
    }
    visited.add(ref);
    const val = cell.value;
    if (val.startsWith('=')) {
      const formula = val.slice(1);
      cell.dataset.formula = formula;
      const expr = formula.replace(REF_REGEX, r => {
        const v = getCell(r).value;
        if (v === 'CIRC') return '0';
        if (v === '') return '0';
        const num = parseFloat(v);
        return isNaN(num) ? '0' : num;
      });
      let result;
      try {
        result = Function('return ' + expr)();
        if (isNaN(result)) result = 0;
      } catch {
        result = 0;
      }
      cell.value = result.toString();
      cell.classList.add('formula');
    } else {
      cell.classList.remove('formula');
      delete cell.dataset.formula;
    }
    visited.delete(ref);
    return parseFloat(cell.value);
  }

  function recalc(ref, visited = new Set()) {
    if (visited.has(ref)) return;
    visited.add(ref);
    evaluateCell(ref, visited);
    const dependents = dependentsMap[ref] || [];
    dependents.forEach(dep => recalc(dep, visited));
    visited.delete(ref);
  }

  function onInput(e) {
    const cell = e.target;
    const ref = cell.dataset.ref;
    updateDependencies(ref);
    recalc(ref);
  }

  document.addEventListener('DOMContentLoaded', () => {
    document.querySelectorAll('.celda').forEach(cell => {
      const ref = cell.dataset.ref;
      cells[ref] = cell;
      if (cell.value.startsWith('=')) cell.dataset.formula = cell.value.slice(1);
      cell.addEventListener('input', onInput);
    });
    Object.keys(cells).forEach(ref => updateDependencies(ref));
    Object.keys(cells).forEach(ref => recalc(ref));
  });
})();
</script>
