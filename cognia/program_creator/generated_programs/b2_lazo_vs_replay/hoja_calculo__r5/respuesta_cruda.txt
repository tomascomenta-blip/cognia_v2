<script>
document.addEventListener('DOMContentLoaded', function(){
  const cells = {};
  const inputs = document.querySelectorAll('.celda');
  inputs.forEach(inp=>{
    cells[inp.dataset.ref] = inp;
    inp.dataset.formula = inp.value;
    inp.addEventListener('input', ()=>{
      inp.dataset.formula = inp.value;
      recalc();
    });
  });

  function evaluateCell(ref, stack){
    if(stack.includes(ref)) return 'CIRC';
    stack.push(ref);
    const cell = cells[ref];
    const formula = cell.dataset.formula;
    if(!formula || !formula.startsWith('=')){
      stack.pop();
      return formula;
    }
    const expr = formula.slice(1);
    const result = evaluateExpression(expr, stack);
    stack.pop();
    return result;
  }

  function evaluateExpression(expr, stack){
    let i=0;
    let result=0;
    let currentOp='+';
    while(i<expr.length){
      if(expr[i]===' ') {i++; continue;}
      if(/[A-Z]/.test(expr[i])){
        const ref = expr.substr(i,2);
        i+=2;
        const val = evaluateCell(ref, stack);
        if(val==='CIRC') return 'CIRC';
        const num = Number(val);
        if(isNaN(num)) return 'CIRC';
        if(currentOp==='+') result+=num;
        else if(currentOp==='-') result-=num;
      }else if(/[0-9]/.test(expr[i])||expr[i]==='.'){
        let numStr='';
        while(i<expr.length && /[0-9.]/.test(expr[i])){
          numStr+=expr[i++];
        }
        const num = Number(numStr);
        if(currentOp==='+') result+=num;
        else if(currentOp==='-') result-=num;
      }else if(expr[i]==='+'||expr[i]==='-'){
        currentOp=expr[i++];
      }else{
        i++;
      }
    }
    return result;
  }

  function recalc(){
    inputs.forEach(inp=>{
      const formula = inp.dataset.formula;
      if(formula && formula.startsWith('=')){
        const res = evaluateCell(inp.dataset.ref, []);
        if(res==='CIRC'){
          inp.value='CIRC';
          inp.classList.remove('computed');
        }else{
          const formatted = Number(res).toLocaleString('en-US',{minimumFractionDigits:0,maximumFractionDigits:2});
          inp.value=formatted;
          inp.classList.add('computed');
        }
      }else{
        inp.value=formula;
        inp.classList.remove('computed');
      }
    });
  }

  document.getElementById('recalcBtn').addEventListener('click', recalc);
  recalc();
});
</script>
