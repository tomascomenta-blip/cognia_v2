<script>
(() => {
  const grid = document.getElementById('grid');
  const counterEl = document.getElementById('counter');
  const estadoEl = document.getElementById('estado');
  const resetBtn = document.getElementById('reset');
  const playBtn = document.getElementById('playagain');

  const mines = new Set([6,12,18]); // fixed mines
  const size = 5;
  const total = size*size;
  const cells = []; // array of cell elements
  const numbers = new Array(total).fill(0); // adjacent mine counts

  // Precompute adjacent mine counts
  for (let i=0;i<total;i++){
    if (mines.has(i)) continue;
    let r = Math.floor(i/size), c = i%size;
    let count = 0;
    for (let dr=-1;dr<=1;dr++){
      for (let dc=-1;dc<=1;dc++){
        if (dr===0 && dc===0) continue;
        let nr=r+dr, nc=c+dc;
        if (nr>=0 && nr<size && nc>=0 && nc<size){
          let idx = nr*size+nc;
          if (mines.has(idx)) count++;
        }
      }
    }
    numbers[i] = count;
  }

  // Create cells
  for (let i=0;i<total;i++){
    const div = document.createElement('div');
    div.className='c';
    div.dataset.i = i;
    div.addEventListener('click', () => handleClick(i));
    div.addEventListener('contextmenu', (e)=>{e.preventDefault(); toggleFlag(i);});
    grid.appendChild(div);
    cells.push(div);
  }

  let openedCount = 0;
  let gameOver = false;

  function handleClick(i){
    if (gameOver) return;
    const cell = cells[i];
    if (cell.classList.contains('abierta') || cell.classList.contains('bandera')) return;
    if (mines.has(i)){
      // Open mine
      cell.classList.add('abierta');
      cell.dataset.mine = 'true';
      cell.textContent = '💣';
      estadoEl.textContent = 'perdiste';
      gameOver = true;
      return;
    }
    openCell(i);
    // If zero, cascade
    if (numbers[i]===0){
      cascadeOpen(i);
    }
  }

  function openCell(i){
    const cell = cells[i];
    if (cell.classList.contains('abierta')) return;
    cell.classList.add('abierta');
    const num = numbers[i];
    cell.dataset.number = num;
    if (num>0) cell.textContent = num;
    openedCount++;
    counterEl.textContent = openedCount;
  }

  function cascadeOpen(start){
    const queue = [start];
    const visited = new Set();
    visited.add(start);
    while(queue.length){
      const idx = queue.shift();
      let r = Math.floor(idx/size), c = idx%size;
      for (let dr=-1;dr<=1;dr++){
        for (let dc=-1;dc<=1;dc++){
          if (dr===0 && dc===0) continue;
          let nr=r+dr, nc=c+dc;
          if (nr>=0 && nr<size && nc>=0 && nc<size){
            let nIdx = nr*size+nc;
            if (visited.has(nIdx)) continue;
            const nCell = cells[nIdx];
            if (nCell.classList.contains('bandera')) continue;
            openCell(nIdx);
            if (numbers[nIdx]===0){
              queue.push(nIdx);
            }
            visited.add(nIdx);
          }
        }
      }
    }
  }

  function toggleFlag(i){
    const cell = cells[i];
    if (cell.classList.contains('abierta')) return;
    if (cell.classList.contains('bandera')){
      cell.classList.remove('bandera');
      cell.textContent = '';
    } else {
      cell.classList.add('bandera');
      cell.textContent = '🚩';
    }
  }

  function resetGame(){
    cells.forEach(cell=>{
      cell.className='c';
      cell.textContent='';
      delete cell.dataset.number;
      delete cell.dataset.mine;
    });
    openedCount = 0;
    counterEl.textContent = openedCount;
    estadoEl.textContent = '';
    gameOver = false;
  }

  resetBtn.addEventListener('click', resetGame);
  playBtn.addEventListener('click', resetGame);
})();
</script>
</body>
