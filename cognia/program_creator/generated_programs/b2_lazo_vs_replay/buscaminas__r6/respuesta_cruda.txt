Title: Buscaminas 5x5  
Description: Un juego de buscaminas 5x5 con 3 minas fijas y lógica de apertura automática.

```html
<!DOCTYPE html>

  }

<script>
/* CONFIGURABLE CONSTANTS */
const GRID_SIZE = 5;
const TOTAL_CELLS = GRID_SIZE * GRID_SIZE;
const MINES = [6,12,18]; // posiciones fijas de las minas

/* UTILIDADES */
function getNeighbors(i){
  const neighbors = [];
  const x = i % GRID_SIZE;
  const y = Math.floor(i / GRID_SIZE);
  for(let dy=-1;dy<=1;dy++){
    for(let dx=-1;dx<=1;dx++){
      if(dx===0 && dy===0) continue;
      const nx = x + dx;
      const ny = y + dy;
      if(nx>=0 && nx<GRID_SIZE && ny>=0 && ny<GRID_SIZE){
        neighbors.push(ny*GRID_SIZE + nx);
      }
    }
  }
  return neighbors;
}

/* CREAR GRID */
const gridEl = document.getElementById('grid');
const estadoEl = document.getElementById('estado');
const cells = [];

for(let i=0;i<TOTAL_CELLS;i++){
  const cell = document.createElement('div');
  cell.classList.add('c');
  cell.dataset.i = i;
  cell.dataset.mine = MINES.includes(i) ? 'true' : 'false';
  // calcular minas adyacentes
  const adj = getNeighbors(i).filter(n => MINES.includes(n)).length;
  cell.dataset.adjacent = adj;
  // eventos
  cell.addEventListener('click', leftClick);
  cell.addEventListener('contextmenu', rightClick);
  gridEl.appendChild(cell);
  cells.push(cell);
}

/* EVENTOS */
function leftClick(e){
  const cell = e.currentTarget;
  if(cell.classList.contains('bandera') || cell.classList.contains('abierta')) return;
  openCell(cell);
}

function rightClick(e){
  e.preventDefault();
  const cell = e.currentTarget;
  if(cell.classList.contains('abierta')) return;
  if(cell.classList.contains('bandera')){
    cell.classList.remove('bandera');
    cell.textContent = '';
  }else{
    cell.classList.add('bandera');
    cell.textContent = '🚩';
  }
}

/* LÓGICA DE ABRIR */
function openCell(cell){
  if(cell.classList.contains('abierta') || cell.classList.contains('bandera')) return;
  if(cell.dataset.mine === 'true'){
    cell.classList.add('abierta');
    cell.textContent = '💣';
    estadoEl.textContent = 'perdiste';
    return;
  }
  cell.classList.add('abierta');
  const adj = parseInt(cell.dataset.adjacent,10);
  if(adj>0) cell.textContent = adj;
  if(adj===0) cascadeOpen(cell);
}

function cascadeOpen(startCell){
  const queue = [startCell];
  const visited = new Set();
  while(queue.length){
    const cell = queue.shift();
    const i = parseInt(cell.dataset.i,10);
    const neighbors = getNeighbors(i);
    for(const n of neighbors){
      const neighborCell = cells[n];
      if(neighborCell.classList.contains('abierta') || neighborCell.classList.contains('bandera')) continue;
      if(neighborCell.dataset.mine === 'true') continue;
      neighborCell.classList.add('abierta');
      const adj = parseInt(neighborCell.dataset.adjacent,10);
      if(adj>0) neighborCell.textContent = adj;
      if(adj===0 && !visited.has(n)){
        visited.add(n);
        queue.push(neighborCell);
      }
    }
  }
}
</script>
```
