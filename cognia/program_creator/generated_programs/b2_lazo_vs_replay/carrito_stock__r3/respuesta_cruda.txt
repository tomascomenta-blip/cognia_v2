<script>
document.addEventListener('DOMContentLoaded', function(){
  const cart = {}; // {id: quantity}
  const linesContainer = document.querySelector('.lines');
  const totalSpan = document.getElementById('total');

  function formatCurrency(value){
    return value.toLocaleString('es-ES', {style:'currency', currency:'EUR'});
  }

  function updateCartUI(){
    // Clear lines
    linesContainer.innerHTML = '';
    // Rebuild lines
    for(const id in cart){
      const qty = cart[id];
      const line = document.createElement('div');
      line.className = 'linea';
      line.dataset.id = id;
      line.innerHTML = `
        <span class="id">${id}</span>
        <span class="cant">${qty}</span>
        <button class="quitar" data-id="${id}">Quitar</button>
      `;
      linesContainer.appendChild(line);
    }
    // Update total
    let total = 0;
    for(const id in cart){
      const prodDiv = document.querySelector(`.prod[data-id="${id}"]`);
      const price = parseFloat(prodDiv.dataset.precio);
      total += price * cart[id];
    }
    totalSpan.textContent = formatCurrency(total);
  }

  function toggleAddButton(id, enable){
    const btn = document.querySelector(`.prod[data-id="${id}"] .add`);
    if(btn){
      btn.disabled = !enable;
    }
  }

  function addToCart(id){
    const prodDiv = document.querySelector(`.prod[data-id="${id}"]`);
    const stock = parseInt(prodDiv.dataset.stock, 10);
    const currentQty = cart[id] || 0;
    if(currentQty < stock){
      cart[id] = currentQty + 1;
      if(cart[id] === stock){
        toggleAddButton(id, false);
      }
      updateCartUI();
    }
  }

  function removeFromCart(id){
    if(!cart[id]) return;
    cart[id] -= 1;
    if(cart[id] <= 0){
      delete cart[id];
      toggleAddButton(id, true);
    } else {
      // If quantity decreased below stock, enable button
      toggleAddButton(id, true);
    }
    updateCartUI();
  }

  // Add button listeners
  document.querySelectorAll('.add').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      addToCart(btn.dataset.id);
    });
  });

  // Remove button delegation
  linesContainer.addEventListener('click', function(e){
    if(e.target.classList.contains('quitar')){
      removeFromCart(e.target.dataset.id);
    }
  });

  // Initial UI
  updateCartUI();
});
</script>
</body>
</html>
