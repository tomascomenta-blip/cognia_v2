<script>
document.addEventListener('DOMContentLoaded', function() {
  const cells = {};
  document.querySelectorAll('.celda').forEach(el => {
    const ref = el.dataset.ref;
    cells[ref] = el;
    el.addEventListener('input', onInput);
  });

  function onInput(e) {
    const input = e.target;
    const val = input.value.trim();
    if (val.startsWith('=')) {
      input.dataset.formula = val;
    } else {
      delete input.dataset.formula;
    }
    recalcAll();
  }

  function recalcAll() {
    Object.keys(cells).forEach(ref => {
      const cell = cells[ref];
      if (cell.dataset.formula) {
        const val = evaluateCell(ref, new Set());
        if (val === 'CIRC') {
          cell.value = 'CIRC';
          cell.parentNode.classList.add('error');
        } else {
          cell.value = val;
          cell.parentNode.classList.remove('error');
        }
      } else {
        cell.parentNode.classList.remove('error');
      }
    });
  }

  function evaluateCell(ref, visited) {
    const cell = cells[ref];
    if (!cell) return 0;
    if (visited.has(ref)) return 'CIRC';
    visited.add(ref);
    const formula = cell.dataset.formula;
    if (!formula) {
      const text = cell.value.trim();
      const num = parseFloat(text);
      visited.delete(ref);
      return isNaN(num) ? 0 : num;
    }
    let expr = formula.slice(1);
    expr = expr.replace(/[A-C][1-3]/g, function(match) {
      const val = evaluateCell(match, visited);
      if (val === 'CIRC') return 'CIRC';
      return val;
    });
    if (expr.includes('CIRC')) {
      visited.delete(ref);
      return 'CIRC';
    }
    let result;
    try {
      result = Function('"use strict";return (' + expr + ')')();
    } catch (e) {
      result = 0;
    }
    visited.delete(ref);
    return result;
  }

  recalcAll();
});
</script>
