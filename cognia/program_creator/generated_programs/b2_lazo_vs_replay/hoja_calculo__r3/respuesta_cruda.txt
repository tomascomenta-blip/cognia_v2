<script>
(() => {
  const cells = Array.from(document.querySelectorAll('.celda'));
  const cellMap = new Map();
  cells.forEach(c => cellMap.set(c.dataset.ref, c));

  const updateTime = () => {
    document.getElementById('time').textContent = new Date().toLocaleTimeString();
  };

  const parseFormula = (formula) => {
    // Extract cell references
    const refs = formula.match(/[A-C][1-3]/g) || [];
    return refs;
  };

  const evaluateCell = (ref, visited = new Set()) => {
    const cell = cellMap.get(ref);
    if (!cell) return 'CIRC';
    const val = cell.value.trim();
    if (!val.startsWith('=')) {
      const num = parseFloat(val);
      return isNaN(num) ? '' : num;
    }
    if (visited.has(ref)) {
      return 'CIRC';
    }
    visited.add(ref);
    const expr = val.slice(1);
    const refs = parseFormula(expr);
    let evalExpr = expr;
    for (const r of refs) {
      const res = evaluateCell(r, visited);
      if (res === 'CIRC') {
        visited.delete(ref);
        return 'CIRC';
      }
      evalExpr = evalExpr.replaceAll(r, res);
    }
    let result;
    try {
      result = Function(`return ${evalExpr}`)();
    } catch {
      result = 'CIRC';
    }
    visited.delete(ref);
    return result;
  };

  const recalcAll = () => {
    cells.forEach(cell => {
      const ref = cell.dataset.ref;
      const val = cell.value.trim();
      if (val.startsWith('=')) {
        const res = evaluateCell(ref);
        cell.value = res === 'CIRC' ? 'CIRC' : res;
      }
    });
    updateTime();
  };

  cells.forEach(cell => {
    cell.addEventListener('input', () => {
      recalcAll();
    });
  });

  document.getElementById('recalcBtn').addEventListener('click', recalcAll);

  // Initial calculation
  recalcAll();
})();
</script>
