Title: Kanban Board  
Description: Un tablero kanban con tres columnas y tarjetas que se mueven entre ellas con reglas de WIP.

```html

<script>
document.addEventListener('DOMContentLoaded', function(){
  const board = document.getElementById('board');
  const todoCol = board.querySelector('[data-col="todo"]');
  const doingCol = board.querySelector('[data-col="doing"]');
  const doneCol = board.querySelector('[data-col="done"]');

  function createCard(id){
    const card = document.createElement('div');
    card.className = 'card';
    card.dataset.id = id;

    const header = document.createElement('div');
    header.className = 'card-header';
    header.textContent = id;

    const btnMas = document.createElement('button');
    btnMas.className = 'mas';
    const btnMen = document.createElement('button');
    btnMen.className = 'menos';

    header.appendChild(btnMas);
    header.appendChild(btnMen);

    const body = document.createElement('div');
    body.className = 'card-body';
    body.textContent = 'Contenido de ' + id;

    card.appendChild(header);
    card.appendChild(body);
    return card;
  }

  ['t1','t2','t3'].forEach(id => {
    todoCol.querySelector('.card-list').appendChild(createCard(id));
  });

  function updateCounters(){
    const todoCount = todoCol.querySelectorAll('.card').length;
    const doingCount = doingCol.querySelectorAll('.card').length;
    const doneCount = doneCol.querySelectorAll('.card').length;

    document.getElementById('cont-todo').textContent = todoCount;
    document.getElementById('cont-doing').textContent = doingCount;
    document.getElementById('cont-done').textContent = doneCount;

    // Enable/disable buttons according to position and WIP rule
    board.querySelectorAll('.card').forEach(card => {
      const col = card.parentElement.parentElement.dataset.col; // card-list's parent
      const masBtn = card.querySelector('.mas');
      const menosBtn = card.querySelector('.menos');

      if(col === 'todo'){
        masBtn.disabled = false;
        menosBtn.disabled = true;
      } else if(col === 'doing'){
        masBtn.disabled = false;
        menosBtn.disabled = false;
      } else if(col === 'done'){
        masBtn.disabled = true;
        menosBtn.disabled = false;
      }

      // WIP rule: if doing has 2 cards, disable mas on todo cards
      if(col === 'todo' && doingCol.querySelectorAll('.card').length >= 2){
        masBtn.disabled = true;
      }
    });
  }

  function moveCard(card, direction){
    const currentCol = card.parentElement.parentElement; // card-list's parent
    const currentData = currentCol.dataset.col;
    let targetCol;

    if(direction === 'right'){
      if(currentData === 'todo'){
        targetCol = doingCol;
      } else if(currentData === 'doing'){
        targetCol = doneCol;
      } else {
        return;
      }
    } else if(direction === 'left'){
      if(currentData === 'doing'){
        targetCol = todoCol;
      } else if(currentData === 'done'){
        targetCol = doingCol;
      } else {
        return;
      }
    }

    // WIP rule: cannot move into doing if it already has 2 cards
    if(targetCol === doingCol && doingCol.querySelectorAll('.card').length >= 2){
      return;
    }

    targetCol.querySelector('.card-list').appendChild(card);
    updateCounters();
  }

  board.addEventListener('click', function(e){
    if(e.target.classList.contains('mas')){
      const card = e.target.closest('.card');
      moveCard(card, 'right');
    } else if(e.target.classList.contains('menos')){
      const card = e.target.closest('.card');
      moveCard(card, 'left');
    }
  });

  updateCounters();
});
</script>
</body>
</html>
```
