<script>
const PRODUCTS = [
  {id:'p1', name:'Producto 1', price:100, stock:2},
  {id:'p2', name:'Producto 2', price:50, stock:3},
  {id:'p3', name:'Producto 3', price:25, stock:1}
];
const cart = {}; // id -> quantity
function updateCart(){
  const linesContainer = document.querySelector('.lines');
  linesContainer.innerHTML = '';
  let total = 0;
  for(const id in cart){
    const qty = cart[id];
    const prodDiv = document.querySelector(`.prod[data-id="${id}"]`);
    const price = parseFloat(prodDiv.dataset.precio);
    total += price * qty;
    const line = document.createElement('div');
    line.className = 'linea';
    line.dataset.id = id;
    const nameSpan = document.createElement('span');
    nameSpan.className = 'name';
    nameSpan.textContent = prodDiv.querySelector('.name').textContent;
    const qtySpan = document.createElement('span');
    qtySpan.className = 'cant';
    qtySpan.textContent = qty;
    const removeBtn = document.createElement('button');
    removeBtn.className = 'quitar';
    removeBtn.dataset.id = id;
    removeBtn.textContent = 'Quitar';
    removeBtn.addEventListener('click', removeFromCart);
    line.appendChild(nameSpan);
    line.appendChild(qtySpan);
    line.appendChild(removeBtn);
    linesContainer.appendChild(line);
  }
  const totalSpan = document.getElementById('total');
  totalSpan.textContent = total.toLocaleString('es-ES', {style:'currency', currency:'EUR'});
  const addButtons = document.querySelectorAll('.add');
  addButtons.forEach(btn=>{
    const id = btn.dataset.id;
    const prodDiv = document.querySelector(`.prod[data-id="${id}"]`);
    const stock = parseInt(prodDiv.dataset.stock,10);
    const qty = cart[id] || 0;
    btn.disabled = qty >= stock;
  });
}
function addToCart(e){
  const id = e.target.dataset.id;
  const prodDiv = document.querySelector(`.prod[data-id="${id}"]`);
  const stock = parseInt(prodDiv.dataset.stock,10);
  const qty = cart[id] || 0;
  if(qty < stock){
    cart[id] = qty + 1;
    updateCart();
  }
}
function removeFromCart(e){
  const id = e.target.dataset.id;
  const qty = cart[id] || 0;
  if(qty > 0){
    cart[id] = qty - 1;
    if(cart[id] === 0) delete cart[id];
    updateCart();
  }
}
document.querySelectorAll('.add').forEach(btn=>btn.addEventListener('click', addToCart));
</script>
