<script>
const cart = {};

function formatMoney(v){return v.toLocaleString('es-ES',{style:'currency',currency:'EUR'});}

function updateCart(){
  const cartItems=document.getElementById('cart-items');
  cartItems.innerHTML='';
  let total=0;
  for(const id in cart){
    const qty=cart[id];
    const prod=document.querySelector(`.prod[data-id="${id}"]`);
    const price=parseFloat(prod.dataset.precio);
    total+=price*qty;
    const line=document.createElement('div');
    line.className='linea';
    line.dataset.id=id;
    const info=document.createElement('div');
    info.className='info';
    info.textContent=prod.querySelector('.info h3').textContent;
    const cant=document.createElement('span');
    cant.className='cant';
    cant.textContent=qty;
    const quitar=document.createElement('button');
    quitar.className='quitar';
    quitar.dataset.id=id;
    quitar.textContent='-';
    quitar.onclick=()=>removeItem(id);
    line.appendChild(info);
    line.appendChild(cant);
    line.appendChild(quitar);
    cartItems.appendChild(line);
  }
  document.getElementById('total').textContent='Total: '+formatMoney(total);
}

function addItem(id){
  const prod=document.querySelector(`.prod[data-id="${id}"]`);
  const stock=parseInt(prod.dataset.stock);
  const qty=cart[id]||0;
  if(qty<stock){
    cart[id]=qty+1;
    if(cart[id]===stock){
      const btn=prod.querySelector('.add');
      btn.disabled=true;
    }
    updateCart();
  }
}

function removeItem(id){
  const prod=document.querySelector(`.prod[data-id="${id}"]`);
  const qty=cart[id]||0;
  if(qty>0){
    cart[id]=qty-1;
    if(cart[id]===0){
      delete cart[id];
      const btn=prod.querySelector('.add');
      btn.disabled=false;
    }else if(cart[id]===parseInt(prod.dataset.stock)-1){
      const btn=prod.querySelector('.add');
      btn.disabled=false;
    }
    updateCart();
  }
}

document.querySelectorAll('.add').forEach(btn=>{
  btn.addEventListener('click',()=>addItem(btn.dataset.id));
});
</script>
