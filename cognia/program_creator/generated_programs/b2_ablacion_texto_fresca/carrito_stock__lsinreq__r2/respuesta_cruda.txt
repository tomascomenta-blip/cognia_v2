Title: Carrito de compra con STOCK  
Description: Una página minimalista con productos y carrito que respeta el stock y actualiza el total en tiempo real.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Carrito de compra con STOCK</title>
<style>
:root{--accent:#008080;--bg:#f5f5f5;--border:#e0e0e0}
body{margin:0;font-family:Arial,Helvetica,sans-serif;background:var(--bg);color:#333}
.container{display:grid;grid-template-columns:1fr 300px;gap:20px;padding:20px}
.prod{background:#fff;border:1px solid var(--border);border-radius:8px;padding:15px;margin-bottom:15px;display:flex;flex-direction:column}
.prod h3{margin:0 0 10px;font-size:1.1em}
.price{font-weight:bold;font-size:1.2em;margin-bottom:5px}
.stock{font-size:.9em;color:var(--accent);background:#e0f7f7;padding:2px 6px;border-radius:4px;align-self:flex-start}
.add{margin-top:auto;padding:8px 12px;background:var(--accent);color:#fff;border:none;border-radius:4px;cursor:pointer;font-size:.9em}
.add:disabled{background:#bbb;cursor:not-allowed}
.cart{position:sticky;top:20px;background:#fff;border:1px solid var(--border);border-radius:8px;padding:15px;height:fit-content}
.cart h2{margin-top:0}
.linea{display:flex;justify-content:space-between;align-items:center;margin-bottom:10px}
.linea span.cant{font-weight:bold}
.quitar{background:none;border:none;color:var(--accent);cursor:pointer;font-size:.9em}
.total{font-weight:bold;font-size:1.2em;margin-top:10px}
@media(max-width:600px){.container{grid-template-columns:1fr}.cart{position:static;margin-top:20px}}
</style>
</head>
<body>
<div class="container">
  <div id="products">
    <div class="prod" data-id="p1" data-precio="100" data-stock="2">
      <h3>Producto 1</h3>
      <div class="price">€100.00</div>
      <span class="stock">Stock: 2</span>
      <button class="add" data-id="p1">Agregar</button>
    </div>
    <div class="prod" data-id="p2" data-precio="50" data-stock="3">
      <h3>Producto 2</h3>
      <div class="price">€50.00</div>
      <span class="stock">Stock: 3</span>
      <button class="add" data-id="p2">Agregar</button>
    </div>
    <div class="prod" data-id="p3" data-precio="25" data-stock="1">
      <h3>Producto 3</h3>
      <div class="price">€25.00</div>
      <span class="stock">Stock: 1</span>
      <button class="add" data-id="p3">Agregar</button>
    </div>
  </div>
  <div id="cart" class="cart">
    <h2>Carrito</h2>
    <div id="cart-lines"></div>
    <div class="total">Total: <span id="total">€0.00</span></div>
  </div>

<script>
const cartLines=document.getElementById('cart-lines');
const totalSpan=document.getElementById('total');
const products=document.querySelectorAll('.prod');
const cart={};

function formatCurrency(v){return v.toLocaleString('es-ES',{style:'currency',currency:'EUR'});}
function updateTotal(){
  let sum=0;
  for(const id in cart){sum+=cart[id].price*cart[id].qty;}
  totalSpan.textContent=formatCurrency(sum);
}
function addToCart(id){
  const prod=document.querySelector(`.prod[data-id="${id}"]`);
  const price=parseFloat(prod.dataset.precio);
  const stock=parseInt(prod.dataset.stock);
  const addBtn=prod.querySelector('.add');
  if(!cart[id]){cart[id]={price,qty:0};}
  if(cart[id].qty<stock){
    cart[id].qty++;
    if(cart[id].qty===stock){addBtn.disabled=true;}
    renderLine(id);
    updateTotal();
  }
}
function removeFromCart(id){
  if(!cart[id])return;
  const prod=document.querySelector(`.prod[data-id="${id}"]`);
  const addBtn=prod.querySelector('.add');
  const stock=parseInt(prod.dataset.stock);
  cart[id].qty--;
  if(cart[id].qty===0){
    delete cart[id];
    removeLine(id);
  }else{
    if(cart[id].qty<stock){addBtn.disabled=false;}
    renderLine(id);
  }
  updateTotal();
}
function renderLine(id){
  const existing=document.querySelector(`.linea[data-id="${id}"]`);
  const qty=cart[id].qty;
  if(existing){existing.querySelector('.cant').textContent=qty;}
  else{
    const line=document.createElement('div');
    line.className='linea';
    line.dataset.id=id;
    line.innerHTML=`<span class="cant">${qty}</span> <button class="quitar" data-id="${id}">Quitar</button>`;
    cartLines.appendChild(line);
  }
}
function removeLine(id){
  const line=document.querySelector(`.linea[data-id="${id}"]`);
  if(line)line.remove();
}
products.forEach(prod=>{
  const btn=prod.querySelector('.add');
  btn.addEventListener('click',()=>addToCart(prod.dataset.id));
});
cartLines.addEventListener('click',e=>{
  if(e.target.classList.contains('quitar')){
    removeFromCart(e.target.dataset.id);
  }
});
</script>
</body>
</html>
```
