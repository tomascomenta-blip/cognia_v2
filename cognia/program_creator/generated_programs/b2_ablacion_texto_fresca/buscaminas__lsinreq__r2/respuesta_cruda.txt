<script>
const mines=[6,12,18];
const cells=document.querySelectorAll('.c');
const counterEl=document.getElementById('counter');
const estadoEl=document.getElementById('estado');
let flags=0;
function adj(i){let r=Math.floor(i/5),c=i%5, cnt=0;
for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
if(dr===0&&dc===0)continue;
let nr=r+dr,nc=c+dc;
if(nr<0||nr>4||nc<0||nc>4)continue;
if(mines.includes(nr*5+nc))cnt++;}return cnt;}
function open(i){let cell=cells[i];
if(cell.classList.contains('abierta')||cell.classList.contains('bandera'))return;
cell.classList.add('abierta');
let n=adj(i);
cell.textContent=n?n:'';
if(mines.includes(i)){
estadoEl.textContent='perdiste';
cells.forEach(c=>{
if(mines.includes(parseInt(c.dataset.i)))c.textContent='*';
});return;}
if(n===0){
for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
if(dr===0&&dc===0)continue;
let nr=Math.floor(i/5)+dr,nc=i%5+dc;
if(nr<0||nr>4||nc<0||nc>4)continue;
open(nr*5+nc);
}}}
cells.forEach((cell,i)=>{
cell.addEventListener('click',()=>open(i));
cell.addEventListener('contextmenu',e=>{
e.preventDefault();
if(cell.classList.contains('abierta'))return;
if(cell.classList.toggle('bandera')){flags++;}else{flags--;}
counterEl.textContent='Bandera: '+flags+'/3';
});
});
document.getElementById('reset').addEventListener('click',()=>{
cells.forEach(c=>{c.classList.remove('abierta','bandera');c.textContent='';});
flags=0;counterEl.textContent='Bandera: 0/3';estadoEl.textContent='';
});
</script>
