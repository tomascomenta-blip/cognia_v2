Title: Kanban en HTML  
Description: Un tablero Kanban con tres columnas y tarjetas que se mueven entre ellas.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Kanban en HTML</title>
<style>
  body{font-family:Arial,Helvetica,sans-serif;margin:0;padding:0;background:#f5f5f5;}
  .board{display:flex;flex-wrap:wrap;justify-content:space-between;padding:10px;}
  .col{flex:1;min-width:250px;background:#fff;margin:5px;padding:10px;border-radius:5px;box-shadow:0 2px 5px rgba(0,0,0,.1);}
  .col h2{margin:0 0 5px;font-size:1.2em;color:#333;}
  .col span{display:block;margin-bottom:10px;font-weight:bold;color:#555;}
  .card{position:relative;background:#e0e0e0;padding:10px;margin-bottom:10px;border-radius:4px;font-size:.9em;}
  .card button{position:absolute;top:5px;border:none;border-radius:50%;width:20px;height:20px;font-size:12px;color:#fff;cursor:pointer;background:#ccc;}
  .card button.menos{right:30px;}
  .card button.mas{right:5px;}
  .card button:disabled{background:#aaa;cursor:not-allowed;}
  .card button:not(:disabled):hover{background:#008080;}
</style>
</head>
<body>
<div class="board"></div>
<script>
const MAX_DOING = 2;
const initialCards = ['t1','t2','t3'];

function createCard(id){
  const card=document.createElement('div');
  card.className='card';
  card.dataset.id=id;
  card.textContent=id;
  const btnMas=document.createElement('button');
  btnMas.className='mas';
  btnMas.textContent='+';
  const btnMenos=document.createElement('button');
  btnMenos.className='menos';
  btnMenos.textContent='-';
  card.appendChild(btnMas);
  card.appendChild(btnMenos);
  return card;
}

function getColumn(card){return card.parentElement.parentElement;}
function getColumnName(col){return col.dataset.col;}

function moveCard(card,direction){
  const col=getColumn(card);
  const colName=getColumnName(col);
  let targetColName;
  if(direction==='right'){
    if(colName==='todo') targetColName='doing';
    else if(colName==='doing') targetColName='done';
    else return;
  }else if(direction==='left'){
    if(colName==='doing') targetColName='todo';
    else if(colName==='done') targetColName='doing';
    else return;
  }
  const targetCol=document.querySelector(`.col[data-col="${targetColName}"]`);
  if(targetColName==='doing' && targetCol.querySelectorAll('.card').length>=MAX_DOING) return;
  targetCol.querySelector('.cards').appendChild(card);
  updateAll();
}

function updateCounters(){
  const todo=document.querySelector('.col[data-col="todo"] .cards').children.length;
  const doing=document.querySelector('.col[data-col="doing"] .cards').children.length;
  const done=document.querySelector('.col[data-col="done"] .cards').children.length;
  document.getElementById('cont-todo').textContent=todo;
  document.getElementById('cont-doing').textContent=doing;
  document.getElementById('cont-done').textContent=done;
}

function updateButtons(card){
  const colName=getColumnName(getColumn(card));
  const btnMas=card.querySelector('.mas');
  const btnMenos=card.querySelector('.menos');
  if(colName==='done') btnMas.disabled=true;
  else if(colName==='doing') btnMas.disabled=false;
  else if(colName==='todo'){
    const doingCol=document.querySelector('.col[data-col="doing"] .cards');
    btnMas.disabled=doingCol.children.length>=MAX_DOING;
  }
  btnMenos.disabled=colName==='todo';
}

function updateAll(){
  updateCounters();
  document.querySelectorAll('.card').forEach(updateButtons);
}

function init(){
  const board=document.querySelector('.board');
  ['todo','doing','done'].forEach(colName=>{
    const col=document.createElement('div');
    col.className='col';
    col.dataset.col=colName;
    const h2=document.createElement('h2');
    h2.textContent=colName.charAt(0).toUpperCase()+colName.slice(1);
    const counter=document.createElement('span');
    counter.id=`cont-${colName}`;
    const cardsDiv=document.createElement('div');
    cardsDiv.className='cards';
    col.appendChild(h2);
    col.appendChild(counter);
    col.appendChild(cardsDiv);
    board.appendChild(col);
  });
  const todoCards=document.querySelector('.col[data-col="todo"] .cards');
  initialCards.forEach(id=>{
    const card=createCard(id);
    todoCards.appendChild(card);
  });
  document.querySelectorAll('.card').forEach(card=>{
    card.querySelector('.mas').addEventListener('click',()=>moveCard(card,'right'));
    card.querySelector('.menos').addEventListener('click',()=>moveCard(card,'left'));
  });
  updateAll();
}
document.addEventListener('DOMContentLoaded',init);
</script>
</body>
</html>
```
