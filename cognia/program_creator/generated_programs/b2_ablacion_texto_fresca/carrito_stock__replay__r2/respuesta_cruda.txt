Title: Carrito con Stock  
Description: Un carrito de compra con control de stock en una sola página HTML.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Carrito con Stock</title>
<style>
:root{--accent:#008080;--bg:#f5f5f5;--card-bg:#fff;--text:#333;--border:#ddd;}
body{margin:0;font-family:Arial,Helvetica,sans-serif;background:var(--bg);color:var(--text);}
.container{display:flex;gap:20px;padding:20px;}
.products{flex:1;display:grid;grid-template-columns:repeat(auto-fit,minmax(200px,1fr));gap:20px;}
.prod{background:var(--card-bg);border:1px solid var(--border);border-radius:8px;padding:15px;display:flex;flex-direction:column;justify-content:space-between;}
.prod .name{font-size:1.1rem;font-weight:600;margin-bottom:10px;}
.prod .price{font-size:1.2rem;font-weight:700;margin-bottom:10px;}
.prod .stock{display:inline-block;background:var(--accent);color:#fff;padding:2px 6px;border-radius:4px;font-size:0.9rem;margin-bottom:10px;}
.prod .add{background:var(--accent);color:#fff;border:none;padding:8px 12px;border-radius:4px;cursor:pointer;font-size:1rem;}
.prod .add:disabled{background:#aaa;cursor:not-allowed;}
.cart{width:300px;background:var(--card-bg);border:1px solid var(--border);border-radius:8px;padding:15px;position:sticky;top:20px;height:fit-content;}
.cart h2{margin:0 0 15px 0;font-size:1.3rem;}
.cart .linea{display:flex;align-items:center;justify-content:space-between;margin-bottom:10px;}
.cart .linea .name{font-weight:600;}
.cart .linea .cant{margin:0 10px;}
.cart .linea .quitar{background:#e74c3c;color:#fff;border:none;padding:4px 8px;border-radius:4px;cursor:pointer;font-size:0.9rem;}
.cart .linea .quitar:hover{background:#c0392b;}
.cart .total{margin-top:20px;font-size:1.2rem;font-weight:700;text-align:right;}
@media (max-width:600px){.container{flex-direction:column;}.cart{width:100%;position:static;}}
</style>
</head>
<body>
<div class="container">
  <div class="products">
    <div class="prod" data-id="p1" data-precio="100" data-stock="2">
      <h3 class="name">Producto 1</h3>
      <p class="price">100,00 €</p>
      <span class="stock">Stock: 2</span>
      <button class="add" data-id="p1">Añadir</button>
    </div>
    <div class="prod" data-id="p2" data-precio="50" data-stock="3">
      <h3 class="name">Producto 2</h3>
      <p class="price">50,00 €</p>
      <span class="stock">Stock: 3</span>
      <button class="add" data-id="p2">Añadir</button>
    </div>
    <div class="prod" data-id="p3" data-precio="25" data-stock="1">
      <h3 class="name">Producto 3</h3>
      <p class="price">25,00 €</p>
      <span class="stock">Stock: 1</span>
      <button class="add" data-id="p3">Añadir</button>
    </div>
  </div>
  <div class="cart">
    <h2>Carrito</h2>
    <div id="cart-items"></div>
    <div class="total">Total: <span id="total">0,00 €</span></div>
  </div>
</div>
<script>
(function(){
  const cart = new Map();
  function updateTotal(){
    let total=0;
    cart.forEach((qty,id)=>{
      const prod=document.querySelector(`.prod[data-id="${id}"]`);
      const price=parseFloat(prod.dataset.precio);
      total+=qty*price;
    });
    document.getElementById('total').textContent=total.toLocaleString('es-ES',{style:'currency',currency:'EUR'});
  }
  function addToCart(id){
    const prod=document.querySelector(`.prod[data-id="${id}"]`);
    const stock=parseInt(prod.dataset.stock);
    const qty=cart.get(id)||0;
    if(qty>=stock)return;
    cart.set(id,qty+1);
    let line=document.querySelector(`.linea[data-id="${id}"]`);
    if(!line){
      line=document.createElement('div');
      line.className='linea';
      line.dataset.id=id;
      line.innerHTML=`<span class="name"></span> <span class="cant"></span> <button class="quitar" data-id="${id}">Quitar</button>`;
      document.getElementById('cart-items').appendChild(line);
      line.querySelector('.quitar').addEventListener('click',()=>removeFromCart(id));
    }
    line.querySelector('.cant').textContent=cart.get(id);
    line.querySelector('.name').textContent=prod.querySelector('.name').textContent;
    const addBtn=document.querySelector(`.add[data-id="${id}"]`);
    if(cart.get(id)>=stock)addBtn.disabled=true;else addBtn.disabled=false;
    updateTotal();
  }
  function removeFromCart(id){
    const qty=cart.get(id)||0;
    if(qty<=0)return;
    const newQty=qty-1;
    if(newQty<=0){
      cart.delete(id);
      const line=document.querySelector(`.linea[data-id="${id}"]`);
      if(line)line.remove();
    }else{
      cart.set(id,newQty);
      const line=document.querySelector(`.linea[data-id="${id}"]`);
      if(line)line.querySelector('.cant').textContent=newQty;
    }
    const prod=document.querySelector(`.prod[data-id="${id}"]`);
    const stock=parseInt(prod.dataset.stock);
    const addBtn=document.querySelector(`.add[data-id="${id}"]`);
    if(newQty<stock)addBtn.disabled=false;
    updateTotal();
  }
  document.querySelectorAll('.add').forEach(btn=>btn.addEventListener('click',()=>addToCart(btn.dataset.id)));
})();
</script>
</body>
</html>
```
