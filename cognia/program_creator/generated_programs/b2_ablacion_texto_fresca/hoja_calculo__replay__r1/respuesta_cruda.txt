Description: Una hoja de cálculo de 3x3 con fórmulas y detección de ciclos.  
HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Mini hoja de cálculo</title>
<style>
body{margin:0;font-family:Arial,Helvetica,sans-serif;background:#f0f0f0;}
#status{position:fixed;top:10px;left:10px;color:#4caf50;font-size:18px;}
#grid{display:grid;grid-template-columns:repeat(3,80px);gap:8px;justify-content:center;margin-top:80px;}
.celda{width:80px;height:30px;border:1px solid #ccc;border-radius:4px;box-shadow:0 1px 3px rgba(0,0,0,.1);background:#fff;font-size:14px;text-align:right;padding:0 4px;outline:none;transition:border-color .2s;}
.celda:focus{border-color:#008080;}
</style>

<div id="status">✓</div>
<div id="grid">
  <input class="celda" data-ref="A1">
  <input class="celda" data-ref="B1" data-formula="=A1+1">
  <input class="celda" data-ref="C1" data-formula="=B1+1">
  <input class="celda" data-ref="A2">
  <input class="celda" data-ref="B2">
  <input class="celda" data-ref="C2">
  <input class="celda" data-ref="A3">
  <input class="celda" data-ref="B3">
  <input class="celda" data-ref="C3">
</div>
<script>
const cells={};
document.querySelectorAll('.celda').forEach(el=>{
  cells[el.dataset.ref]=el;
  el.addEventListener('input',()=>{if(el.value.startsWith('='))el.dataset.formula=el.value;else delete el.dataset.formula;evaluateAll();});
});
function evalCell(ref,stack){
  if(stack.includes(ref))return'CIRC';
  const el=cells[ref];
  const f=el.dataset.formula;
  if(!f){const v=el.value;const n=parseFloat(v);return isNaN(n)?v:n;}
  stack.push(ref);
  const expr=f.slice(1).replace(/[A-C][1-3]/g,m=>{const r=evalCell(m,stack);return r==='CIRC'?'NaN':r;});
  let res;
  try{res=Function('return '+expr)();}
  catch{res='CIRC';}
  stack.pop();
  return res;
}
function evaluateAll(){
  Object.keys(cells).forEach(ref=>{
    const val=evalCell(ref,[]);
    cells[ref].value=val==='CIRC'?val:val;
  });
}
evaluateAll();
</script>
</body>
</html>
```
