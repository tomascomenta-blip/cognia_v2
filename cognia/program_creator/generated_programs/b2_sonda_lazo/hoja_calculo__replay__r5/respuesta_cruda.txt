<script>
const cells = {};
document.querySelectorAll('.celda').forEach(el=>{
  cells[el.dataset.ref] = el;
  el.addEventListener('input',()=>{handleInput(el);});
});
document.getElementById('recalc').addEventListener('click',recomputeAll);

function handleInput(el){
  const val = el.value.trim();
  if(val.startsWith('=')){
    el.dataset.formula = val;
    el.dataset.raw = '';
  }else{
    el.dataset.formula = '';
    el.dataset.raw = val;
  }
  recomputeAll();
}

function recomputeAll(){
  Object.keys(cells).forEach(ref=>{evaluateCell(ref,[]);});
}

function evaluateCell(ref, stack){
  const el = cells[ref];
  const formula = el.dataset.formula;
  if(!formula || !formula.startsWith('=')){
    el.value = el.dataset.raw;
    el.classList.remove('formula','circular');
    return {value:el.dataset.raw, circular:false};
  }
  if(stack.includes(ref)){
    el.value = 'CIRC';
    el.classList.add('circular');
    el.classList.remove('formula');
    return {value:'CIRC', circular:true};
  }
  const expr = formula.slice(1);
  const refs = expr.match(/[A-C][1-3]/g) || [];
  let evalExpr = expr;
  for(const r of refs){
    const res = evaluateCell(r, stack.concat(ref));
    if(res.circular){
      el.value = 'CIRC';
      el.classList.add('circular');
      el.classList.remove('formula');
      return {value:'CIRC', circular:true};
    }
    const num = parseFloat(res.value);
    const val = isNaN(num)?0:num;
    evalExpr = evalExpr.replaceAll(r, val);
  }
  try{
    const result = eval(evalExpr);
    const formatted = Number.isFinite(result)
      ? result.toLocaleString('es-ES',{minimumFractionDigits:2,maximumFractionDigits:2})
      : 'CIRC';
    el.value = formatted;
    el.classList.add('formula');
    el.classList.remove('circular');
    return {value:formatted, circular:false};
  }catch(e){
    el.value = 'CIRC';
    el.classList.add('circular');
    el.classList.remove('formula');
    return {value:'CIRC', circular:true};
  }
}
recomputeAll();
</script>
