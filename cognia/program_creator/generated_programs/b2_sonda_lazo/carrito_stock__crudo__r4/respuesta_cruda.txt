<script>
document.addEventListener('DOMContentLoaded', function(){
  const cart = {}; // id -> quantity
  const linesContainer = document.getElementById('lines');
  const totalSpan = document.getElementById('total');

  function updateTotal(){
    let total = 0;
    for(const id in cart){
      const qty = cart[id];
      const price = parseFloat(document.querySelector(`.prod[data-id="${id}"]`).dataset.precio);
      total += qty * price;
    }
    totalSpan.textContent = total;
  }

  function renderLine(id){
    const qty = cart[id];
    const line = document.createElement('div');
    line.className = 'linea';
    line.dataset.id = id;
    line.innerHTML = `<span class="cant">${qty}</span> x <span class="name">${id}</span> <button class="quitar" data-id="${id}">Quitar</button>`;
    linesContainer.appendChild(line);
  }

  function updateLine(id){
    const line = document.querySelector(`.linea[data-id="${id}"]`);
    if(line){
      line.querySelector('.cant').textContent = cart[id];
    }
  }

  function removeLine(id){
    const line = document.querySelector(`.linea[data-id="${id}"]`);
    if(line) line.remove();
  }

  function enableAdd(id){
    const btn = document.querySelector(`.add[data-id="${id}"]`);
    if(btn) btn.disabled = false;
  }

  function disableAdd(id){
    const btn = document.querySelector(`.add[data-id="${id}"]`);
    if(btn) btn.disabled = true;
  }

  function addProduct(id){
    const prod = document.querySelector(`.prod[data-id="${id}"]`);
    const stock = parseInt(prod.dataset.stock);
    const current = cart[id] || 0;
    if(current < stock){
      cart[id] = current + 1;
      if(current === 0){
        renderLine(id);
      } else {
        updateLine(id);
      }
      if(cart[id] === stock){
        disableAdd(id);
      }
      updateTotal();
    }
  }

  function removeProduct(id){
    const current = cart[id] || 0;
    if(current > 0){
      cart[id] = current - 1;
      if(cart[id] === 0){
        removeLine(id);
        delete cart[id];
        enableAdd(id);
      } else {
        updateLine(id);
      }
      updateTotal();
    }
  }

  document.querySelectorAll('.add').forEach(btn=>{
    btn.addEventListener('click',()=>addProduct(btn.dataset.id));
  });

  document.addEventListener('click',function(e){
    if(e.target.classList.contains('quitar')){
      removeProduct(e.target.dataset.id);
    }
  });
});
</script>
