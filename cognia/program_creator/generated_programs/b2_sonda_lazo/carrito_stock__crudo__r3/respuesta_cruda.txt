<script>
  const cart = {}; // {id: quantity}
  const products = document.querySelectorAll('.prod');

  // Helper to get product data
  function getProduct(id){
    const prod = document.querySelector(`.prod[data-id="${id}"]`);
    return {
      price: parseInt(prod.dataset.precio,10),
      stock: parseInt(prod.dataset.stock,10)
    };
  }

  // Update cart UI
  function updateCart(){
    const cartDiv = document.getElementById('cart');
    cartDiv.innerHTML = '';
    let total = 0;
    for(const id in cart){
      const qty = cart[id];
      const {price} = getProduct(id);
      total += qty * price;

      const line = document.createElement('div');
      line.className = 'linea';
      line.dataset.id = id;

      const span = document.createElement('span');
      span.className = 'cant';
      span.textContent = `${id} x ${qty} = $${qty*price}`;

      const btn = document.createElement('button');
      btn.className = 'quitar';
      btn.dataset.id = id;
      btn.textContent = 'Quitar';

      line.appendChild(span);
      line.appendChild(btn);
      cartDiv.appendChild(line);
    }
    document.getElementById('total').textContent = total;
  }

  // Enable/disable add button based on stock
  function setAddButtonState(id, disabled){
    const btn = document.querySelector(`.add[data-id="${id}"]`);
    btn.disabled = disabled;
  }

  // Add button handler
  function addHandler(e){
    const id = e.target.dataset.id;
    const {stock} = getProduct(id);
    const current = cart[id] || 0;
    if(current < stock){
      cart[id] = current + 1;
      updateCart();
      if(cart[id] === stock) setAddButtonState(id,true);
    }
  }

  // Quitar button handler
  function quitarHandler(e){
    const id = e.target.dataset.id;
    const current = cart[id];
    if(current > 1){
      cart[id] = current - 1;
      updateCart();
      setAddButtonState(id,false);
    }else{
      delete cart[id];
      updateCart();
      setAddButtonState(id,false);
    }
  }

  // Attach event listeners
  products.forEach(prod=>{
    const addBtn = prod.querySelector('.add');
    addBtn.addEventListener('click',addHandler);
  });

  document.getElementById('cart').addEventListener('click',function(e){
    if(e.target.classList.contains('quitar')){
      quitarHandler(e);
    }
  });
</script>
