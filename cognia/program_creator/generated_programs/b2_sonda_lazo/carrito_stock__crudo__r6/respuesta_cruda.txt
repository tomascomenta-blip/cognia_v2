<script>
  // Cart data structure: {id: quantity}
  const cart = {};

  // Helper to get product element by id
  function getProdEl(id){ return document.querySelector(`.prod[data-id="${id}"]`); }

  // Update total display
  function updateTotal(){
    let total = 0;
    for(const id in cart){
      const prod = getProdEl(id);
      const price = parseFloat(prod.dataset.precio);
      total += cart[id] * price;
    }
    document.getElementById('total').textContent = total.toFixed(2);
  }

  // Create or update a line in the cart
  function renderLine(id){
    const linesDiv = document.getElementById('lines');
    let line = linesDiv.querySelector(`.linea[data-id="${id}"]`);
    const qty = cart[id];
    if(qty > 0){
      if(line){
        line.querySelector('.cant').textContent = qty;
      }else{
        line = document.createElement('div');
        line.className = 'linea';
        line.dataset.id = id;
        line.innerHTML = `
          <span>${id}</span>
          <span class="cant">${qty}</span>
          <button class="quitar" data-id="${id}">-</button>
        `;
        linesDiv.appendChild(line);
        // Attach quitar listener
        line.querySelector('.quitar').addEventListener('click', () => removeItem(id));
      }
    }else{
      if(line) line.remove();
    }
  }

  // Add item to cart
  function addItem(id){
    const prod = getProdEl(id);
    const stock = parseInt(prod.dataset.stock);
    const currentQty = cart[id] || 0;
    if(currentQty < stock){
      cart[id] = currentQty + 1;
      renderLine(id);
      updateTotal();
      if(cart[id] === stock){
        prod.querySelector('.add').disabled = true;
      }
    }
  }

  // Remove item from cart
  function removeItem(id){
    const prod = getProdEl(id);
    const currentQty = cart[id] || 0;
    if(currentQty > 0){
      cart[id] = currentQty - 1;
      renderLine(id);
      updateTotal();
      if(cart[id] < parseInt(prod.dataset.stock)){
        prod.querySelector('.add').disabled = false;
      }
    }
  }

  // Attach listeners to add buttons
  document.querySelectorAll('.add').forEach(btn=>{
    btn.addEventListener('click', () => addItem(btn.dataset.id));
  });

  // Initial animation on load
  window.addEventListener('load', () => {
    // Fade in the entire container
    document.querySelector('.container').style.opacity = 0;
    setTimeout(()=>{ document.querySelector('.container').style.transition='opacity .5s'; document.querySelector('.container').style.opacity=1; }, 100);
  });
</script>
