Title: Carrito con Stock  
Description: Un carrito de compra con control de stock y actualización dinámica.

<script>
const products=document.querySelectorAll('.prod');const cartItems=document.getElementById('cart-items');const totalSpan=document.getElementById('total');const cart={};
function updateTotal(){let sum=0;for(const id in cart){sum+=cart[id].qty*cart[id].price;}totalSpan.textContent=sum.toLocaleString('es-ES',{style:'currency',currency:'EUR'});}
function addToCart(id){const prod=document.querySelector(`.prod[data-id="${id}"]`);const price=parseFloat(prod.dataset.precio);const stock=parseInt(prod.dataset.stock);if(!cart[id]){cart[id]={price,qty:0};}if(cart[id].qty<stock){cart[id].qty++;if(cart[id].qty===stock){prod.querySelector('.add').disabled=true;}}else{return;}renderCart();updateTotal();}
function removeFromCart(id){if(cart[id]){cart[id].qty--;if(cart[id].qty===0){delete cart[id];const prod=document.querySelector(`.prod[data-id="${id}"]`);prod.querySelector('.add').disabled=false;}else{const prod=document.querySelector(`.prod[data-id="${id}"]`);if(cart[id].qty<parseInt(prod.dataset.stock))prod.querySelector('.add').disabled=false;}}renderCart();updateTotal();}
function renderCart(){cartItems.innerHTML='';for(const id in cart){const line=document.createElement('div');line.className='linea';line.dataset.id=id;const span=document.createElement('span');span.className='cant';span.textContent=`${cart[id].qty} x ${cart[id].price.toLocaleString('es-ES',{style:'currency',currency:'EUR'})}`;const btn=document.createElement('button');btn.className='quitar';btn.dataset.id=id;btn.textContent='-';btn.onclick=()=>removeFromCart(id);line.appendChild(span);line.appendChild(btn);cartItems.appendChild(line);} }
products.forEach(p=>p.querySelector('.add').addEventListener('click',e=>addToCart(e.target.dataset.id)));
</script>
</body>
