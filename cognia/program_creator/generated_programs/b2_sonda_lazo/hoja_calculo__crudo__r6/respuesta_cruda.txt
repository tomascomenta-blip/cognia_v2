<script>
document.addEventListener('DOMContentLoaded', function(){
  const inputs = document.querySelectorAll('.celda');
  const cells = {};

  function updateDependencies(cell){
    cell.dependencies.forEach(ref=>{
      const dep = cells[ref];
      if(dep) dep.dependents = dep.dependents.filter(c=>c!==cell);
    });
    cell.dependencies = [];
    if(cell.formula){
      const refs = cell.formula.match(/\b([A-C][1-3])\b/g) || [];
      refs.forEach(ref=>{
        if(!cells[ref]) return;
        cell.dependencies.push(ref);
        cells[ref].dependents.push(cell);
      });
    }
  }

  function computeCell(cell, stack){
    if(stack.has(cell.ref)){
      cell.element.value = 'CIRC';
      return null;
    }
    stack.add(cell.ref);
    let val;
    if(cell.formula){
      const expr = cell.formula;
      const refs = expr.match(/\b([A-C][1-3])\b/g) || [];
      let evalExpr = expr;
      for(const ref of refs){
        const refCell = cells[ref];
        const refVal = computeCell(refCell, stack);
        if(refVal===null){
          cell.element.value = 'CIRC';
          stack.delete(cell.ref);
          return null;
        }
        evalExpr = evalExpr.replace(new RegExp(ref,'g'), refVal);
      }
      try{ val = eval(evalExpr); }catch(e){ val = 0; }
      cell.element.value = val.toString();
    }else{
      const raw = cell.element.value;
      val = parseInt(raw,10);
      if(isNaN(val)) val = 0;
    }
    stack.delete(cell.ref);
    cell.dependents.forEach(dep=>computeCell(dep, stack));
    return val;
  }

  function onInput(e){
    const inp = e.target;
    const ref = inp.dataset.ref;
    const cell = cells[ref];
    const txt = inp.value;
    if(txt.startsWith('=')){
      cell.formula = txt.slice(1);
    }else{
      cell.formula = null;
    }
    updateDependencies(cell);
    computeCell(cell, new Set());
  }

  inputs.forEach(inp=>{
    const ref = inp.dataset.ref;
    const cell = {ref, element:inp, formula:null, dependents:[], dependencies:[]};
    cells[ref] = cell;
    if(inp.value.startsWith('=')) cell.formula = inp.value.slice(1);
  });

  Object.values(cells).forEach(c=>updateDependencies(c));
  Object.values(cells).forEach(c=>computeCell(c, new Set()));
  inputs.forEach(inp=>inp.addEventListener('input', onInput));
});
</script>
