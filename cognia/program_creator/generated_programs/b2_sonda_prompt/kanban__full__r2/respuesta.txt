Title: Tablero Kanban  
Description: Un tablero Kanban con tres columnas y tarjetas que se mueven entre ellas con límites de WIP.  
HTML Code:  
```html
<!DOCTYPE html>
<html lang="es"><head><meta charset="utf-8"><title>Tablero Kanban</title><style>
body{font-family:sans-serif;margin:0;padding:0;display:flex;flex-direction:column;align-items:center;}
.board{display:flex;gap:10px;width:100%;max-width:900px;padding:10px;}
.col{flex:1;background:#f0f0f0;padding:10px;border-radius:5px;}
.col h2{margin:0 0 10px;font-size:1.2em;}
.card{background:white;padding:5px;margin-bottom:5px;border-radius:3px;display:flex;justify-content:space-between;align-items:center;}
.card button{margin-left:2px;}
</style></head><body>
<div class="board">
<div class="col" data-col="todo"><h2>Pendiente <span id="cont-todo"></span></h2></div>
<div class="col" data-col="doing"><h2>En Progreso <span id="cont-doing"></span></h2></div>
<div class="col" data-col="done"><h2>Hecho <span id="cont-done"></span></h2></div>
</div>
<script>
function c(id){var d=document.createElement('div');d.className='card';d.dataset.id=id;d.textContent='Tarea '+id;
var m=document.createElement('button');m.className='mas';m.textContent='→';
var n=document.createElement('button');n.className='menos';n.textContent='←';
d.appendChild(m);d.appendChild(n);return d;}
function u(){['todo','doing','done'].forEach(c=>{var n=document.querySelectorAll('.col[data-col="'+c+'"] .card').length;
document.getElementById('cont-'+c).textContent=n;});}
function m(card,dir){var col=card.parentElement.dataset.col,cols=['todo','doing','done'],i=cols.indexOf(col),j=i+dir;
if(j<0||j>2)return;if(cols[j]=='doing'){var d=document.querySelectorAll('.col[data-col="doing"] .card');if(d.length>=2)return;}
document.querySelector('.col[data-col="'+cols[j]+'"]').appendChild(card);u();}
document.addEventListener('DOMContentLoaded',()=>{['t1','t2','t3'].forEach(i=>{document.querySelector('.col[data-col="todo"]').appendChild(c(i));});u();
document.querySelectorAll('.col').forEach(col=>{col.addEventListener('click',e=>{if(e.target.classList.contains('mas'))m(e.target.parentElement,1);else if(e.target.classList.contains('menos'))m(e.target.parentElement,-1);});});});
</script>
</body></html>
```
