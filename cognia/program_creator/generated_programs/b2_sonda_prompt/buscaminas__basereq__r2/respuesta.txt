<script>
document.addEventListener('DOMContentLoaded', () => {
  const mines = [6,12,18];
  const estado = document.getElementById('estado');
  const cells = Array.from(document.querySelectorAll('.c'));
  const neighborCounts = new Array(25).fill(0);

  const getNeighbors = (i) => {
    const neighbors = [];
    const r = Math.floor(i/5);
    const c = i%5;
    for(let dr=-1; dr<=1; dr++){
      for(let dc=-1; dc<=1; dc++){
        if(dr===0 && dc===0) continue;
        const nr=r+dr, nc=c+dc;
        if(nr>=0 && nr<5 && nc>=0 && nc<5){
          neighbors.push(nr*5+nc);
        }
      }
    }
    return neighbors;
  };

  for(let i=0;i<25;i++){
    const neighbors = getNeighbors(i);
    neighborCounts[i] = neighbors.filter(n=>mines.includes(n)).length;
  }

  const openCell = (cell, idx) => {
    if(cell.classList.contains('abierta') || cell.classList.contains('bandera')) return;
    if(mines.includes(idx)){
      cell.classList.add('abierta');
      cell.textContent = '💣';
      estado.textContent = 'perdiste';
      return;
    }
    cell.classList.add('abierta');
    const count = neighborCounts[idx];
    cell.textContent = count>0?count:'';
    if(count===0){
      cascadeOpen(idx);
    }
  };

  const cascadeOpen = (startIdx) => {
    const queue = [startIdx];
    const visited = new Set();
    while(queue.length){
      const idx = queue.shift();
      const cell = cells[idx];
      if(cell.classList.contains('abierta') || cell.classList.contains('bandera')) continue;
      if(mines.includes(idx)) continue;
      cell.classList.add('abierta');
      const count = neighborCounts[idx];
      cell.textContent = count>0?count:'';
      if(count===0){
        const neighbors = getNeighbors(idx);
        for(const n of neighbors){
          if(!visited.has(n)){
            visited.add(n);
            queue.push(n);
          }
        }
      }
    }
  };

  cells.forEach(cell=>{
    const idx = parseInt(cell.dataset.i,10);
    cell.addEventListener('click', () => openCell(cell, idx));
    cell.addEventListener('contextmenu', (e)=>{
      e.preventDefault();
      if(cell.classList.contains('abierta')) return;
      if(cell.classList.contains('bandera')){
        cell.classList.remove('bandera');
        cell.textContent='';
      }else{
        cell.classList.add('bandera');
        cell.textContent='⚑';
      }
    });
  });
});
</script>
