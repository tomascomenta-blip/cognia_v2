Title: Mini hoja de cálculo  
Description: Una mini hoja de cálculo 3x3 con fórmulas y detección de ciclos.  
HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Mini hoja de cálculo</title>
<style>
  body{font-family:Arial,Helvetica,sans-serif;margin:0;padding:20px;background:#f9f9f9;}
  h1{font-size:1.5rem;text-align:center;margin-bottom:1rem;}
  table{border-collapse:collapse;margin:auto;width:100%;max-width:400px;}
  td{border:1px solid #ccc;}
  .celda{
    width:80px;height:30px;text-align:right;padding:5px;font-size:1rem;
    box-sizing:border-box;
  }
  .celda.circ{color:red;}
</style>
</head>
<body>
<h1>Mini hoja de cálculo</h1>
<table>
  <tr>
    <td><input type="text" class="celda" data-ref="A1"></td>
    <td><input type="text" class="celda" data-ref="A2"></td>
    <td><input type="text" class="celda" data-ref="A3"></td>
  </tr>
  <tr>
    <td><input type="text" class="celda" data-ref="B1"></td>
    <td><input type="text" class="celda" data-ref="B2"></td>
    <td><input type="text" class="celda" data-ref="B3"></td>
  </tr>
  <tr>
    <td><input type="text" class="celda" data-ref="C1"></td>
    <td><input type="text" class="celda" data-ref="C2"></td>
    <td><input type="text" class="celda" data-ref="C3"></td>
  </tr>
</table>

<script>
(() => {
  const cells = {}; // ref -> input element
  const refs = ['A1','A2','A3','B1','B2','B3','C1','C2','C3'];
  refs.forEach(r => cells[r] = document.querySelector(`input[data-ref="${r}"]`));

  let ignoreInput = false;

  // Helper to format numbers
  function formatNumber(n) {
    return n.toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2});
  }

  // Evaluate a cell recursively
  function evaluateCell(ref, stack) {
    const el = cells[ref];
    if (!el) return 0;
    if (stack.has(ref)) {
      // Circular reference
      el.value = 'CIRC';
      el.classList.add('circ');
      return 'CIRC';
    }
    if (!el.dataset.formula) {
      // Not a formula, return numeric value
      const val = parseFloat(el.value);
      return isNaN(val) ? 0 : val;
    }
    // Formula cell
    stack.add(ref);
    const expr = el.dataset.formula.slice(1); // remove leading '='
    const parts = expr.split(/([+-])/);
    let result = evaluateOperand(parts[0], stack);
    if (result === 'CIRC') {
      el.value = 'CIRC';
      el.classList.add('circ');
      stack.delete(ref);
      return 'CIRC';
    }
    for (let i = 1; i < parts.length; i += 2) {
      const op = parts[i];
      const operand = evaluateOperand(parts[i+1], stack);
      if (operand === 'CIRC') {
        el.value = 'CIRC';
        el.classList.add('circ');
        stack.delete(ref);
        return 'CIRC';
      }
      if (op === '+') result += operand;
      else result -= operand;
    }
    // Set formatted value
    el.value = formatNumber(result);
    el.classList.remove('circ');
    stack.delete(ref);
    return result;
  }

  function evaluateOperand(token, stack) {
    token = token.trim();
    if (/^[A-C][1-3]$/.test(token)) {
      const val = evaluateCell(token, stack);
      return val === 'CIRC' ? 'CIRC' : val;
    }
    const num = parseFloat(token);
    return isNaN(num) ? 0 : num;
  }

  // Recompute all cells
  function recomputeAll() {
    ignoreInput = true;
    refs.forEach(r => evaluateCell(r, new Set()));
    ignoreInput = false;
  }

  // Initialize formulas from initial values
  refs.forEach(r => {
    const el = cells[r];
    if (el.value.startsWith('=')) {
      el.dataset.formula = el.value;
    }
  });

  // Input event handler
  refs.forEach(r => {
    const el = cells[r];
    el.addEventListener('input', () => {
      if (ignoreInput) return;
      if (el.value.startsWith('=')) {
        el.dataset.formula = el.value;
      } else {
        delete el.dataset.formula;
      }
      recomputeAll();
    });
  });

  // Initial calculation
  recomputeAll();
})();
</script>
</body>
</html>
```
