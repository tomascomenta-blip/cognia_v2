<script>
(function(){
  const cells = document.querySelectorAll('input.celda');
  const refMap = {};
  cells.forEach(cell=>{
    refMap[cell.dataset.ref] = cell;
    cell.dataset.formula = cell.value;
  });

  let isUpdating = false;

  function computeAll(){
    if(isUpdating) return;
    isUpdating = true;
    const cache = {};

    function computeValue(ref, stack){
      if(stack.has(ref)) return 'CIRC';
      if(cache[ref] !== undefined) return cache[ref];
      const cell = refMap[ref];
      if(!cell) return '0';
      const formula = cell.dataset.formula;
      if(!formula.startsWith('=')){
        cache[ref] = formula;
        return formula;
      }
      stack.add(ref);
      let expr = formula.slice(1);
      let cycle = false;
      expr = expr.replace(/\b([A-C][1-3])\b/g, (m, r)=>{
        const val = computeValue(r, stack);
        if(val === 'CIRC'){ cycle = true; return '0'; }
        const num = parseFloat(val);
        return isNaN(num) ? '0' : num;
      });
      if(cycle){ cache[ref] = 'CIRC'; stack.delete(ref); return 'CIRC'; }
      let result;
      try{ result = eval(expr); }catch(e){ result = 0; }
      cache[ref] = result.toString();
      stack.delete(ref);
      return cache[ref];
    }

    cells.forEach(cell=>{
      const ref = cell.dataset.ref;
      const val = computeValue(ref, new Set());
      if(cell.dataset.formula.startsWith('=')){
        cell.value = val;
      }else{
        cell.value = cell.dataset.formula;
      }
    });

    isUpdating = false;
  }

  cells.forEach(cell=>{
    cell.addEventListener('input', function(){
      if(isUpdating) return;
      this.dataset.formula = this.value;
      computeAll();
    });
  });

  computeAll();
})();
</script>
