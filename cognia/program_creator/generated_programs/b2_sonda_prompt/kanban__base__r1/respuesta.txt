<script>
document.addEventListener('DOMContentLoaded', function() {
  function updateCounters() {
    const todoCount = document.querySelectorAll('.col[data-col="todo"] .card').length;
    const doingCount = document.querySelectorAll('.col[data-col="doing"] .card').length;
    const doneCount = document.querySelectorAll('.col[data-col="done"] .card').length;
    document.getElementById('cont-todo').textContent = todoCount;
    document.getElementById('cont-doing').textContent = doingCount;
    document.getElementById('cont-done').textContent = doneCount;
  }
  function updateButtons() {
    const doingCount = document.querySelectorAll('.col[data-col="doing"] .card').length;
    const cards = document.querySelectorAll('.card');
    cards.forEach(function(card) {
      const col = card.parentElement.dataset.col;
      const masBtn = card.querySelector('.mas');
      const menosBtn = card.querySelector('.menos');
      if(col === 'todo') {
        masBtn.disabled = doingCount >= 2;
        menosBtn.disabled = true;
      } else if(col === 'doing') {
        masBtn.disabled = false;
        menosBtn.disabled = false;
      } else if(col === 'done') {
        masBtn.disabled = true;
        menosBtn.disabled = false;
      }
    });
  }
  function moveCard(card, targetCol) {
    const targetColumn = document.querySelector('.col[data-col="' + targetCol + '"]');
    targetColumn.appendChild(card);
    updateCounters();
    updateButtons();
  }
  function handleMas(btn) {
    const card = btn.closest('.card');
    const currentCol = card.parentElement.dataset.col;
    if(currentCol === 'todo') {
      const doingCount = document.querySelectorAll('.col[data-col="doing"] .card').length;
      if(doingCount >= 2) return;
      moveCard(card, 'doing');
    } else if(currentCol === 'doing') {
      moveCard(card, 'done');
    }
  }
  function handleMenos(btn) {
    const card = btn.closest('.card');
    const currentCol = card.parentElement.dataset.col;
    if(currentCol === 'doing') {
      moveCard(card, 'todo');
    } else if(currentCol === 'done') {
      moveCard(card, 'doing');
    }
  }
  updateCounters();
  updateButtons();
  document.addEventListener('click', function(e) {
    if(e.target.classList.contains('mas')) {
      handleMas(e.target);
    } else if(e.target.classList.contains('menos')) {
      handleMenos(e.target);
    }
  });
});
</script>
</body>
</html>
