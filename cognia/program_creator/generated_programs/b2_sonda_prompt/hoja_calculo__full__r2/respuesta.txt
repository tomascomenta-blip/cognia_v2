Title: Mini Hoja de Cálculo 3x3  
Description: Una hoja de cálculo simple con fórmulas y detección de ciclos.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="UTF-8">
<title>Mini Hoja de Cálculo</title>
<style>
  body{font-family:sans-serif;margin:0;padding:1rem;background:#f0f0f0;}
  .grid{display:grid;grid-template-columns:repeat(3,1fr);gap:0.5rem;}
  .celda{width:100%;padding:0.5rem;font-size:1rem;text-align:right;}
</style>
</head>
<body>
<h2>Mini Hoja de Cálculo 3x3</h2>
<div class="grid">
  <input class="celda" data-ref="A1" placeholder="A1">
  <input class="celda" data-ref="A2" placeholder="A2">
  <input class="celda" data-ref="A3" placeholder="A3">
  <input class="celda" data-ref="B1" placeholder="B1">
  <input class="celda" data-ref="B2" placeholder="B2">
  <input class="celda" data-ref="B3" placeholder="B3">
  <input class="celda" data-ref="C1" placeholder="C1">
  <input class="celda" data-ref="C2" placeholder="C2">
  <input class="celda" data-ref="C3" placeholder="C3">
</div>
<script>
const cells=document.querySelectorAll('.celda');
const refMap={};
cells.forEach(c=>refMap[c.dataset.ref]=c);
function evalCell(cell,stack=[]) {
  if(stack.includes(cell.dataset.ref)) return 'CIRC';
  const formula=cell.dataset.formula;
  if(!formula) return cell.value;
  const expr=formula.slice(1).replace(/[A-C][1-3]/g,ref=>{
    const r=refMap[ref];
    const val=evalCell(r,stack.concat(cell.dataset.ref));
    if(val==='CIRC') return 'CIRC';
    return parseFloat(val)||0;
  });
  if(expr.includes('CIRC')) return 'CIRC';
  try{const res=eval(expr);return res.toString();}catch{return 'CIRC';}
}
function recalcAll(){
  cells.forEach(c=>{
    const val=evalCell(c);
    c.value=val;
  });
}
cells.forEach(c=>{
  c.addEventListener('input',e=>{
    const v=e.target.value;
    if(v.startsWith('=')){
      c.dataset.formula=v;
    }else{
      delete c.dataset.formula;
    }
    recalcAll();
  });
});
recalcAll();
</script>
</body>
</html>
```
