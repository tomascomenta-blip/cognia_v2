Title: Mini hoja de cálculo 3x3  
Description: Una mini hoja de cálculo con fórmulas y detección de referencias circulares.  
HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="UTF-8">
<title>Mini hoja de cálculo</title>
<style>
table{border-collapse:collapse;width:100%;}
td{border:1px solid #ccc;padding:0;}
input{width:100%;box-sizing:border-box;border:none;padding:8px;font-size:1rem;}
</style>
</head>
<body>
<table>
<tr><td><input class="celda" data-ref="A1"></td><td><input class="celda" data-ref="A2"></td><td><input class="celda" data-ref="A3"></td></tr>
<tr><td><input class="celda" data-ref="B1"></td><td><input class="celda" data-ref="B2"></td><td><input class="celda" data-ref="B3"></td></tr>
<tr><td><input class="celda" data-ref="C1"></td><td><input class="celda" data-ref="C2"></td><td><input class="celda" data-ref="C3" value="=A1+A2"></td></tr>
</table>
<script>
document.addEventListener('DOMContentLoaded',()=>{const cells=document.querySelectorAll('.celda');const map={};cells.forEach(c=>{map[c.dataset.ref]=c;c.dataset.raw=c.value;c.addEventListener('input',e=>{c.dataset.raw=c.value;if(c.value.startsWith('='))c.dataset.formula=c.value;else delete c.dataset.formula;updateAll();});});function updateAll(){cells.forEach(c=>{c.value=compute(c.dataset.ref,new Set());});}function compute(ref,stack){if(stack.has(ref))return'CIRC';stack.add(ref);const c=map[ref];let raw=c.dataset.raw;let val;if(c.dataset.formula){let expr=c.dataset.formula.slice(1);expr=expr.replace(/[A-C][1-3]/g,match=>{const r=compute(match,stack);if(r==='CIRC')return r;if(r==='')return'(0)';return'('+r+')';});if(expr.includes('CIRC')){stack.delete(ref);return'CIRC';}try{val=Function('return '+expr)();}catch{val='CIRC';}stack.delete(ref);return typeof val==='number'?val.toString():val;}else{val=raw;stack.delete(ref);return val;}}updateAll();});
</script>
</body>
</html>
```
