Title: Mini Hoja de Cálculo 3x3  
Description: Una hoja de cálculo de 3×3 con fórmulas y detección de referencias circulares.  
HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="UTF-8">
<title>Mini Hoja de Cálculo 3x3</title>
<style>
  body{font-family:Arial,sans-serif;margin:0;padding:1rem;background:#f9f9f9;}
  .grid{display:grid;grid-template-columns:repeat(3,1fr);gap:0.5rem;max-width:400px;margin:auto;}
  .celda{width:100%;padding:0.5rem;font-size:1rem;text-align:right;}
</style>
</head>
<body>
<h2 style="text-align:center;">Mini Hoja de Cálculo 3×3</h2>
<div class="grid">
  <input class="celda" data-ref="A1" placeholder="A1">
  <input class="celda" data-ref="A2" placeholder="A2">
  <input class="celda" data-ref="A3" placeholder="A3">
  <input class="celda" data-ref="B1" placeholder="B1">
  <input class="celda" data-ref="B2" placeholder="B2">
  <input class="celda" data-ref="B3" placeholder="B3">
  <input class="celda" data-ref="C1" placeholder="C1">
  <input class="celda" data-ref="C2" placeholder="C2">
  <input class="celda" data-ref="C3" placeholder="C3">
</div>
<script>
(() => {
  const cells = document.querySelectorAll('.celda');
  const refMap = {};
  cells.forEach(c => refMap[c.dataset.ref] = c);

  function isFormula(val){return typeof val==='string' && val.startsWith('=');}

  function computeCell(ref, stack=[]) {
    const cell = refMap[ref];
    if (!cell) return 0;
    const val = cell.value.trim();
    if (!isFormula(val)) return parseFloat(val)||0;
    if (stack.includes(ref)) return 'CIRC';
    stack.push(ref);
    const expr = val.slice(1).replace(/[A-C][1-3]/g, r=>{
      const res = computeCell(r, stack.slice());
      return res==='CIRC'?'CIRC':res;
    });
    if (expr.includes('CIRC')) return 'CIRC';
    try {
      const result = Function('return '+expr)();
      return isNaN(result)?0:result;
    } catch(e){return 0;}
  }

  function recalcAll(){
    cells.forEach(c=>{
      const val = c.value.trim();
      if (isFormula(val)){
        const res = computeCell(c.dataset.ref);
        c.value = res==='CIRC'?'CIRC':res;
      }
    });
  }

  cells.forEach(c=>c.addEventListener('input', recalcAll));
})();
</script>
</body>
</html>
```
