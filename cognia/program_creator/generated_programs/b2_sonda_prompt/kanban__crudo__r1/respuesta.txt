<script>
document.addEventListener('DOMContentLoaded', () => {
  const todoCol = document.querySelector('.col[data-col="todo"] .cards');
  const doingCol = document.querySelector('.col[data-col="doing"] .cards');
  const doneCol  = document.querySelector('.col[data-col="done"] .cards');

  // Crear las 3 tarjetas iniciales en Todo
  for(let i=1;i<=3;i++){
    const card = createCard(`t${i}`);
    todoCol.appendChild(card);
  }

  updateCounts();

  // Función para crear una tarjeta
  function createCard(id){
    const card = document.createElement('div');
    card.className = 'card';
    card.dataset.id = id;
    card.innerHTML = `
      <span>${id}</span>
      <div>
        <button class="mas">→</button>
        <button class="menos">←</button>
      </div>
    `;
    // Eventos
    card.querySelector('.mas').addEventListener('click', () => moveRight(card));
    card.querySelector('.menos').addEventListener('click', () => moveLeft(card));
    return card;
  }

  // Mover tarjeta a la derecha
  function moveRight(card){
    const col = card.closest('.col');
    const current = col.dataset.col;
    if(current === 'done') return; // no puede avanzar

    const targetColName = current === 'todo' ? 'doing' : 'done';
    const targetCol = document.querySelector(`.col[data-col="${targetColName}"] .cards`);

    // Regla WIP: Doing solo 2 tarjetas
    if(targetColName === 'doing' && targetCol.children.length >= 2) return;

    targetCol.appendChild(card);
    updateCounts();
  }

  // Mover tarjeta a la izquierda
  function moveLeft(card){
    const col = card.closest('.col');
    const current = col.dataset.col;
    if(current === 'todo') return; // no puede retroceder

    const targetColName = current === 'doing' ? 'todo' : 'doing';
    const targetCol = document.querySelector(`.col[data-col="${targetColName}"] .cards`);

    targetCol.appendChild(card);
    updateCounts();
  }

  // Actualizar contadores
  function updateCounts(){
    document.getElementById('cont-todo').textContent = document.querySelector('.col[data-col="todo"] .cards').children.length;
    document.getElementById('cont-doing').textContent = document.querySelector('.col[data-col="doing"] .cards').children.length;
    document.getElementById('cont-done').textContent = document.querySelector('.col[data-col="done"] .cards').children.length;
  }
});
</script>
