Title: Un carrito de compra con STOCK  
Description: Una página con 3 productos, carrito y control de stock sin usar recursos externos.  

HTML Code:
```html
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="UTF-8">
<title>Carrito con Stock</title>
<style>
  body{font-family:Arial,Helvetica,sans-serif;margin:0;padding:0;background:#f5f5f5;}
  .container{max-width:800px;margin:auto;padding:20px;}
  .products,.cart{background:#fff;padding:15px;border-radius:5px;margin-bottom:20px;}
  .prod{display:flex;justify-content:space-between;align-items:center;margin-bottom:10px;}
  .prod button.add{padding:5px 10px;background:#28a745;color:#fff;border:none;border-radius:3px;cursor:pointer;}
  .prod button.add:disabled{background:#6c757d;cursor:not-allowed;}
  .cart h2{margin-top:0;}
  .linea{display:flex;justify-content:space-between;align-items:center;margin-bottom:8px;}
  .linea button.quitar{padding:3px 6px;background:#dc3545;color:#fff;border:none;border-radius:3px;cursor:pointer;}
  #total{font-weight:bold;font-size:1.2em;}
  @media(max-width:600px){
    .prod{flex-direction:column;align-items:flex-start;}
    .prod button.add{margin-top:5px;}
  }
</style>
</head>
<body>
<div class="container">
  <div class="products">
    <h2>Productos</h2>
    <div class="prod" data-id="p1" data-precio="100" data-stock="2">
      <span>Producto 1 - €100</span>
      <button class="add" data-id="p1">Añadir</button>
    </div>
    <div class="prod" data-id="p2" data-precio="50" data-stock="3">
      <span>Producto 2 - €50</span>
      <button class="add" data-id="p2">Añadir</button>
    </div>
    <div class="prod" data-id="p3" data-precio="25" data-stock="1">
      <span>Producto 3 - €25</span>
      <button class="add" data-id="p3">Añadir</button>
    </div>
  </div>

  <div class="cart">
    <h2>Carrito</h2>
    <div id="lines"></div>
    <div>Total: <span id="total">€0.00</span></div>
  </div>
</div>

<script>
(() => {
  const cart = {}; // {id: qty}
  const products = {}; // {id: {precio, stock, addBtn}}
  const linesContainer = document.getElementById('lines');
  const totalEl = document.getElementById('total');

  // Initialize products
  document.querySelectorAll('.prod').forEach(prodEl => {
    const id = prodEl.dataset.id;
    const precio = parseFloat(prodEl.dataset.precio);
    const stock = parseInt(prodEl.dataset.stock, 10);
    const addBtn = prodEl.querySelector('.add');
    products[id] = {precio, stock, addBtn};

    addBtn.addEventListener('click', () => addToCart(id));
  });

  function addToCart(id) {
    const prod = products[id];
    const qty = cart[id] || 0;
    if (qty < prod.stock) {
      cart[id] = qty + 1;
      updateLine(id);
      if (cart[id] === prod.stock) prod.addBtn.disabled = true;
      updateTotal();
    }
  }

  function removeFromCart(id) {
    if (!cart[id]) return;
    cart[id] -= 1;
    if (cart[id] <= 0) {
      delete cart[id];
      removeLine(id);
      products[id].addBtn.disabled = false;
    } else {
      updateLine(id);
    }
    updateTotal();
  }

  function updateLine(id) {
    const line = document.querySelector(`.linea[data-id="${id}"]`);
    if (line) {
      line.querySelector('.cant').textContent = cart[id];
    } else {
      const prod = products[id];
      const lineEl = document.createElement('div');
      lineEl.className = 'linea';
      lineEl.dataset.id = id;
      lineEl.innerHTML = `
        <span>${id} - €${prod.precio.toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2})}</span>
        <span class="cant">${cart[id]}</span>
        <button class="quitar" data-id="${id}">Quitar</button>
      `;
      lineEl.querySelector('.quitar').addEventListener('click', () => removeFromCart(id));
      linesContainer.appendChild(lineEl);
    }
  }

  function removeLine(id) {
    const line = document.querySelector(`.linea[data-id="${id}"]`);
    if (line) line.remove();
  }

  function updateTotal() {
    let sum = 0;
    for (const id in cart) {
      sum += cart[id] * products[id].precio;
    }
    totalEl.textContent = sum.toLocaleString('es-ES', {style:'currency', currency:'EUR'});
  }

  // Initial render
  updateTotal();
})();
</script>
</body>
</html>
```
