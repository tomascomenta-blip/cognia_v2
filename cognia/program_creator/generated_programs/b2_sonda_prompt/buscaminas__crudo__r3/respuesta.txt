<script>
  const mines = new Set([6,12,18]); // posiciones fijas
  const estado = document.getElementById('estado');

  // devuelve los índices de las celdas vecinas
  function getNeighbors(i){
    const neighbors=[];
    const row=Math.floor(i/5), col=i%5;
    for(let dr=-1; dr<=1; dr++){
      for(let dc=-1; dc<=1; dc++){
        if(dr===0 && dc===0) continue;
        const r=row+dr, c=col+dc;
        if(r>=0 && r<5 && c>=0 && c<5){
          neighbors.push(r*5+c);
        }
      }
    }
    return neighbors;
  }

  // cuenta minas adyacentes
  function countAdjMines(i){
    return getNeighbors(i).filter(n=>mines.has(n)).length;
  }

  // abre una celda
  function openCell(i){
    const cell=document.querySelector(`.c[data-i="${i}"]`);
    if(cell.classList.contains('abierta') || cell.classList.contains('bandera')) return;
    cell.classList.add('abierta');
    const count=countAdjMines(i);
    if(count>0){
      cell.textContent=count;
    }
    if(mines.has(i)){
      estado.textContent='perdiste';
      cell.textContent='💣';
      // opcional: revelar todas las minas
      mines.forEach(m=>{
        const mcell=document.querySelector(`.c[data-i="${m}"]`);
        if(!mcell.classList.contains('abierta')){
          mcell.classList.add('abierta');
          mcell.textContent='💣';
        }
      });
      return;
    }
    if(count===0){
      // cascada
      getNeighbors(i).forEach(n=>openCell(n));
    }
  }

  // eventos de las celdas
  document.querySelectorAll('.c').forEach(cell=>{
    const i=parseInt(cell.dataset.i);
    cell.addEventListener('click',()=>openCell(i));
    cell.addEventListener('contextmenu',e=>{
      e.preventDefault();
      if(cell.classList.contains('abierta')) return;
      cell.classList.toggle('bandera');
    });
  });
</script>
