<script>
document.addEventListener('DOMContentLoaded', function(){
  const cells = {};
  document.querySelectorAll('.celda').forEach(el=>{
    cells[el.dataset.ref] = el;
    el.addEventListener('input', ()=>updateCell(el.dataset.ref));
  });

  function updateCell(ref){
    const el = cells[ref];
    const val = el.value.trim();
    if(val.startsWith('=')){
      el.dataset.formula = val;
    }else{
      delete el.dataset.formula;
    }
    recomputeAll();
  }

  function recomputeAll(){
    const memo = {};
    function evalCell(ref, stack){
      if(memo[ref]) return memo[ref];
      const el = cells[ref];
      const formula = el.dataset.formula;
      if(!formula){
        const num = parseFloat(el.value);
        memo[ref] = {value: isNaN(num)?0:num, error:false};
        return memo[ref];
      }
      if(stack.has(ref)){
        memo[ref] = {value:'CIRC', error:true};
        return memo[ref];
      }
      stack.add(ref);
      let expr = formula.slice(1); // remove leading '='
      const refRegex = /\b([A-C][1-3])\b/g;
      expr = expr.replace(refRegex, (m, r)=>{
        const res = evalCell(r, stack);
        if(res.error) return 'NaN';
        return res.value;
      });
      let val;
      try{
        val = Function('return '+expr)();
        if(typeof val!=='number' || !isFinite(val)) throw new Error();
      }catch{
        val = 'CIRC';
        memo[ref] = {value:val, error:true};
        stack.delete(ref);
        return memo[ref];
      }
      memo[ref] = {value:val, error:false};
      stack.delete(ref);
      return memo[ref];
    }

    Object.keys(cells).forEach(ref=>{
      const el = cells[ref];
      if(el.dataset.formula){
        const res = evalCell(ref, new Set());
        el.value = res.error ? 'CIRC' : res.value;
      }
    });
  }

  // initial calculation
  Object.keys(cells).forEach(ref=>updateCell(ref));
});
</script>
